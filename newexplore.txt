SET NOCOUNT ON;

/* ============================================================
   CWI Q1 2026 — MAJOR GL SCOPE CHECK

   Compare legacy CRE dollars against RAW SAP dollars for the
   biggest discrepancy GLs.

   RAW SAP is intentionally NOT filtered by the current selected
   GL list.

   NO TEMP TABLES.
   ============================================================ */

;WITH CandidateGL AS (
    SELECT GL
    FROM (VALUES
        (4300625), -- VarRent CAM
        (4301049), -- Depreciation - Machinery & Equipment
        (4300652), -- Physical Security
        (4301101), -- Light, Power, Water, Heat
        (4300395), -- Plant Rearrangement Expense
        (4301103), -- Purchased Gas
        (7000301), -- Indirect Labor (Fac Pool A)
        (8320479), -- P&C Property
        (4300805)  -- General Services Ex
    ) X(GL)
),

FacilityAlias AS (
    SELECT *
    FROM (VALUES
        ('ONE AVIATION WAY',          '25167 ONE AVIATION WAY'),
        ('301 VOYAGER WAY NW',        '301 VOYAGER WAY NORTH WEST'),
        ('4065 COLONEL GLENN',        '4065 COLONEL GLENN HWY'),
        ('STRAWBERRY POINT',          '2828 STRAWBERRY POINT RD'),
        ('FAC WR PARK PLC',           '200 PARK PLACE DR'),
        ('FACILITY - WR IGNICO',      '98 IGNICO DR'),
        ('2300 DISCOVERY DR FL',      '2300 DISCOVERY DR'),
        ('8220 SAN PEDRO DR',         '8220 SAN PEDRO SUITE 300'),
        ('75 WEST ST WALPOLE',        '75 WEST ST')
    ) A (OLD_ADDRESS_KEY, NEW_ADDRESS_KEY)
),

LegacyBase AS (
    SELECT
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address])))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS RAW_FACILITY,

        TRY_CONVERT(
            bigint,
            TRY_CONVERT(
                decimal(38,6),
                NULLIF(
                    REPLACE(
                        LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                        ',',
                        ''
                    ),
                    ''
                )
            )
        ) AS GL,

        LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category])))
            AS COST_CATEGORY,

        LTRIM(RTRIM(CONVERT(nvarchar(500), [Cost Element Description])))
            AS GL_DESCRIPTION,

        TRY_CONVERT(decimal(38,2), [Cost])
            AS COST

    FROM ecosystem_source.qmi.controllable_costs

    WHERE TRY_CONVERT(int, [Year]) = 2026
      AND UPPER(
            LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))
          ) = 'Q1'
),

Legacy AS (
    SELECT
        COALESCE(A.NEW_ADDRESS_KEY, L.RAW_FACILITY)
            AS FACILITY,

        L.GL,
        L.COST_CATEGORY,
        L.GL_DESCRIPTION,
        L.COST

    FROM LegacyBase L

    LEFT JOIN FacilityAlias A
      ON A.OLD_ADDRESS_KEY = L.RAW_FACILITY
),

NewExtractFacilities AS (
    SELECT DISTINCT
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), facility)))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS FACILITY

    FROM ecosystem_source.qmi.controllable_costs_new_extract

    WHERE [year] = 2026
      AND [month] BETWEEN 1 AND 3
      AND division = 'DS C2 & Weapons Integration'
),

LegacyFacilities AS (
    SELECT DISTINCT FACILITY
    FROM Legacy
),

CommonFacilities AS (
    SELECT L.FACILITY

    FROM LegacyFacilities L

    JOIN NewExtractFacilities N
      ON N.FACILITY = L.FACILITY
),

CurrentHierarchy AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER)))
            AS COST_CENTER,

        ROW_NUMBER() OVER (
            PARTITION BY UPPER(LTRIM(RTRIM(COST_CENTER)))
            ORDER BY
                last_modified_date DESC,
                created_date DESC,
                id DESC
        ) AS RN

    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy

    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
      AND LTRIM(RTRIM(LEV03_DESC))
            = 'DS C2 & Weapons Integration'
),

FacilityKey AS (
    SELECT
        UPPER(LTRIM(RTRIM([Cost Center])))
            AS COST_CENTER,

        MAX(NULLIF(LTRIM(RTRIM([Address])), ''))
            AS ADDRESS

    FROM ecosystem_source.qmi.costcenterkey

    WHERE NULLIF(LTRIM(RTRIM([Cost Center])), '') IS NOT NULL

    GROUP BY
        UPPER(LTRIM(RTRIM([Cost Center])))
),

RawSAP AS (
    SELECT
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), K.ADDRESS)))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS FACILITY,

        TRY_CONVERT(bigint, T.RACCT)
            AS GL,

        NULLIF(LTRIM(RTRIM(T.GL_TXT20)), '')
            AS GL_DESCRIPTION,

        TRY_CONVERT(decimal(38,2), T.KSL)
            AS COST

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 T

    JOIN CurrentHierarchy H
      ON H.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))
     AND H.RN = 1

    JOIN FacilityKey K
      ON K.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))

    JOIN CandidateGL C
      ON C.GL = TRY_CONVERT(bigint, T.RACCT)

    WHERE TRY_CONVERT(int, T.GJAHR) = 2026
      AND TRY_CONVERT(int, T.POPER) BETWEEN 1 AND 3
      AND TRY_CONVERT(decimal(38,2), T.KSL) IS NOT NULL
),

LegacySummary AS (
    SELECT
        L.GL,

        MAX(L.COST_CATEGORY)
            AS LEGACY_CATEGORY,

        MAX(NULLIF(L.GL_DESCRIPTION, ''))
            AS LEGACY_DESCRIPTION,

        COUNT(DISTINCT L.FACILITY)
            AS LEGACY_FACILITY_COUNT,

        SUM(L.COST)
            AS LEGACY_COST

    FROM Legacy L

    JOIN CandidateGL C
      ON C.GL = L.GL

    JOIN CommonFacilities F
      ON F.FACILITY = L.FACILITY

    GROUP BY
        L.GL
),

SAPSummary AS (
    SELECT
        S.GL,

        MAX(NULLIF(S.GL_DESCRIPTION, ''))
            AS SAP_DESCRIPTION,

        COUNT(DISTINCT S.FACILITY)
            AS SAP_FACILITY_COUNT,

        SUM(S.COST)
            AS RAW_SAP_COST

    FROM RawSAP S

    JOIN CommonFacilities F
      ON F.FACILITY = S.FACILITY

    GROUP BY
        S.GL
)

SELECT
    C.GL,

    L.LEGACY_CATEGORY,

    L.LEGACY_DESCRIPTION,
    S.SAP_DESCRIPTION,

    COALESCE(L.LEGACY_FACILITY_COUNT, 0)
        AS LEGACY_FACILITY_COUNT,

    COALESCE(S.SAP_FACILITY_COUNT, 0)
        AS SAP_FACILITY_COUNT,

    COALESCE(L.LEGACY_COST, 0)
        AS LEGACY_COST,

    COALESCE(S.RAW_SAP_COST, 0)
        AS RAW_SAP_COST,

    COALESCE(S.RAW_SAP_COST, 0)
      - COALESCE(L.LEGACY_COST, 0)
        AS DIFFERENCE

FROM CandidateGL C

LEFT JOIN LegacySummary L
  ON L.GL = C.GL

LEFT JOIN SAPSummary S
  ON S.GL = C.GL

ORDER BY
    ABS(
        COALESCE(S.RAW_SAP_COST, 0)
      - COALESCE(L.LEGACY_COST, 0)
    ) DESC;
