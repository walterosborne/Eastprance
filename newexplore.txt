WITH KeyCenters AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS CC
    FROM ecosystem_source.qmi.costcenterkey
    WHERE NULLIF(LTRIM(RTRIM([Cost Center])), '') IS NOT NULL
),

Transactions AS (
    SELECT
        'ERP' AS Source,
        UPPER(LTRIM(RTRIM(e.[Cost Center]))) AS CC,
        TRY_CONVERT(DECIMAL(19,2), e.[Actual Amount]) AS Amount

    FROM ecosystem_source.dbo.erpdata e

    WHERE e.[Fiscal Year Period] IN (
        '001/2026',
        '002/2026',
        '003/2026'
    )

    AND EXISTS (
        SELECT 1
        FROM KeyCenters k
        WHERE k.CC =
            UPPER(LTRIM(RTRIM(e.[Cost Center])))
    )

    UNION ALL

    SELECT
        'SAP' AS Source,
        UPPER(LTRIM(RTRIM(s.RCNTR))) AS CC,
        TRY_CONVERT(DECIMAL(19,2), s.KSL) AS Amount

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 s

    WHERE TRY_CONVERT(INT, s.GJAHR) = 2026
      AND TRY_CONVERT(INT, s.POPER) BETWEEN 1 AND 3

      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE k.CC =
              UPPER(LTRIM(RTRIM(s.RCNTR)))
      )
)

SELECT
    Source,
    COUNT(*) AS Row_Count,
    COUNT(DISTINCT CC) AS Cost_Centers,

    SUM(CASE
        WHEN Amount > 0 THEN Amount
        ELSE 0
    END) AS Positive_Amount,

    SUM(CASE
        WHEN Amount < 0 THEN ABS(Amount)
        ELSE 0
    END) AS Negative_Absolute,

    SUM(Amount) AS Net_Amount,

    SUM(ABS(Amount)) AS Gross_Activity

FROM Transactions

GROUP BY Source
ORDER BY Source;
