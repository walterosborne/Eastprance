-- Check charge-field coverage for nonzero Gross Labor Timecard hours.
WITH gross_labor AS (
  SELECT
    NULLIF(TRIM(PACCASTY), '') AS paccasty,
    CASE
      WHEN AUFNR IS NULL OR TRIM(AUFNR) RLIKE '^0*$' THEN 'Blank or zero'
      ELSE 'Populated'
    END AS aufnr_status,
    TRY_CAST(NET_MSL AS DECIMAL(24, 4)) AS hours_quantity
  FROM source_systems_dev.sql_central_integration.lfc_transaction_details_s3
  WHERE TRY_CAST(GJAHR AS INT) = 2026
    AND UPPER(TRIM(RUNIT)) = 'H'
    AND TRIM(RACCT) = '0004100000'
)
SELECT
  paccasty,
  aufnr_status,
  COUNT(*) AS postings,
  SUM(hours_quantity) AS signed_hours,
  SUM(ABS(hours_quantity)) AS absolute_hours
FROM gross_labor
WHERE hours_quantity <> 0
GROUP BY paccasty, aufnr_status
ORDER BY absolute_hours DESC, paccasty, aufnr_status;
