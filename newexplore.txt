WITH source_bu AS (
    SELECT LTRIM(RTRIM([Business Unit])) AS BU FROM [qmi].[otd]
    UNION SELECT LTRIM(RTRIM(business_unit)) FROM [qmi].[controllable_costs_new_extract]
    UNION SELECT LTRIM(RTRIM([Business Unit])) FROM [qmi].[controllable_costs_hana]
    UNION SELECT LTRIM(RTRIM([Business Unit])) FROM [qmi].[labor_agg]
    UNION SELECT LTRIM(RTRIM(LEV04_DESC))
    FROM [DTO_Business_Management].[rpt].[rb_load_cost_center_hierarchy]
    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
)
SELECT s.BU, COALESCE(c.[Name], m.[Name]) AS [BU Match]
FROM source_bu s
LEFT JOIN [qmi].[businessunits] c ON c.[Name] = s.BU
LEFT JOIN [qmi].[bugroupings] g ON g.[Original] = s.BU
LEFT JOIN [qmi].[businessunits] m ON m.[Id] = g.[BusinessUnitId]
WHERE NULLIF(s.BU, '') IS NOT NULL
ORDER BY s.BU;
