WITH KeyCenters AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS CC
    FROM ecosystem_source.qmi.costcenterkey
),

Transactions AS (
    SELECT
        'ERP' AS Source,
        TRY_CONVERT(
            INT,
            LEFT([Fiscal Year Period], 3)
        ) AS Month,
        TRY_CONVERT(BIGINT, [G/L Account]) AS GL,
        TRY_CONVERT(
            DECIMAL(19,2),
            [Actual Amount]
        ) AS Amount
    FROM ecosystem_source.dbo.erpdata e
    WHERE [Fiscal Year Period] IN (
        '001/2026', '002/2026', '003/2026'
    )
      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE k.CC =
              UPPER(LTRIM(RTRIM(e.[Cost Center])))
      )

    UNION ALL

    SELECT
        'SAP' AS Source,
        TRY_CONVERT(INT, POPER) AS Month,
        TRY_CONVERT(BIGINT, RACCT) AS GL,
        TRY_CONVERT(DECIMAL(19,2), KSL) AS Amount
    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 s
    WHERE TRY_CONVERT(INT, GJAHR) = 2026
      AND TRY_CONVERT(INT, POPER) BETWEEN 1 AND 3
      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE k.CC =
              UPPER(LTRIM(RTRIM(s.RCNTR)))
      )
)

SELECT
    CASE
        WHEN GROUPING(Month) = 1 THEN 'Q1 TOTAL'
        ELSE CONCAT('Month ', Month)
    END AS Period,

    SUM(CASE
        WHEN Source = 'ERP'
        THEN ABS(Amount)
        ELSE 0
    END) AS ERP_Gross_All,

    SUM(CASE
        WHEN Source = 'ERP'
         AND GL = 8999999
        THEN ABS(Amount)
        ELSE 0
    END) AS ERP_8999999_Gross,

    SUM(CASE
        WHEN Source = 'ERP'
         AND (GL <> 8999999 OR GL IS NULL)
        THEN ABS(Amount)
        ELSE 0
    END) AS ERP_Gross_Excluding_8999999,

    SUM(CASE
        WHEN Source = 'SAP'
         AND (GL <> 8999999 OR GL IS NULL)
        THEN ABS(Amount)
        ELSE 0
    END) AS SAP_Gross_Excluding_8999999,

    SUM(CASE
        WHEN Source = 'ERP'
         AND (GL <> 8999999 OR GL IS NULL)
        THEN Amount
        ELSE 0
    END) AS ERP_Net_Excluding_8999999,

    SUM(CASE
        WHEN Source = 'SAP'
         AND (GL <> 8999999 OR GL IS NULL)
        THEN Amount
        ELSE 0
    END) AS SAP_Net_Excluding_8999999

FROM Transactions

GROUP BY GROUPING SETS (
    (Month),
    ()
)

ORDER BY GROUPING(Month), Month;
