SELECT
    TRY_CONVERT(INT, GJAHR) AS [Year],
    TRY_CONVERT(INT, POPER) AS [Month],

    COUNT(*) AS Row_Count,

    COUNT(DISTINCT RCNTR) AS Cost_Center_Count,
    COUNT(DISTINCT RACCT) AS GL_Account_Count,

    SUM(TRY_CONVERT(DECIMAL(19,2), KSL)) AS Net_Amount,

    SUM(ABS(TRY_CONVERT(DECIMAL(19,2), KSL)))
        AS Gross_Activity

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03

WHERE TRY_CONVERT(INT, GJAHR) >= 2025

GROUP BY
    TRY_CONVERT(INT, GJAHR),
    TRY_CONVERT(INT, POPER)

ORDER BY
    [Year],
    [Month];
