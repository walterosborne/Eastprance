WITH bus AS (
    SELECT LTRIM(RTRIM(LEV04_DESC)) AS bu,
           ROW_NUMBER() OVER (
               PARTITION BY UPPER(LTRIM(RTRIM(COST_CENTER)))
               ORDER BY last_modified_date DESC, created_date DESC, id DESC
           ) AS rn
    FROM [DTO_Business_Management].[rpt].[rb_load_cost_center_hierarchy]
    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
)
SELECT DISTINCT b.bu AS [BU], COALESCE(canonical.[Name], mapped.[Name]) AS [BU Match]
FROM bus b
LEFT JOIN [ecosystem_source].[qmi].[businessunits] canonical ON canonical.[Name] = b.bu
LEFT JOIN [ecosystem_source].[qmi].[bugroupings] g ON g.[Original] = b.bu
LEFT JOIN [ecosystem_source].[qmi].[businessunits] mapped ON mapped.[Id] = g.[BusinessUnitId]
WHERE b.rn = 1 AND NULLIF(b.bu, '') IS NOT NULL
ORDER BY [BU];
