SELECT
    RACCT AS GL_Account,
    GL_TXT20 AS GL_Description,
    ACCT_LEVEL02_TEXT,
    ACCT_LEVEL03_TEXT,

    COUNT(*) AS Row_Count,

    SUM(
        CASE
            WHEN TRY_CONVERT(DECIMAL(19,2), KSL) > 0
            THEN TRY_CONVERT(DECIMAL(19,2), KSL)
            ELSE 0
        END
    ) AS Positive_Amount,

    SUM(
        CASE
            WHEN TRY_CONVERT(DECIMAL(19,2), KSL) < 0
            THEN TRY_CONVERT(DECIMAL(19,2), KSL)
            ELSE 0
        END
    ) AS Negative_Amount,

    SUM(TRY_CONVERT(DECIMAL(19,2), KSL))
        AS Net_Amount

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03

WHERE LTRIM(RTRIM(RCNTR)) = 'TDA84LF0'
  AND TRY_CONVERT(INT, GJAHR) = 2026
  AND TRY_CONVERT(INT, POPER) = 8
  AND TRY_CONVERT(BIGINT, RACCT) IN (
      8320585,
      4300591
  )

GROUP BY
    RACCT,
    GL_TXT20,
    ACCT_LEVEL02_TEXT,
    ACCT_LEVEL03_TEXT

ORDER BY
    RACCT;
