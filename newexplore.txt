SELECT
    LTRIM(RTRIM(t.RCNTR)) AS Cost_Center,

    MAX(h.CC_DESCRIPTION) AS Cost_Center_Name,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT)
             IN (4300591, 4300625)
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Rent,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT)
             IN (4301050, 4301049, 4301048)
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Depreciation_LHI,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT) = 4301102
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Electricity,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT) = 4300703
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Janitorial

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

LEFT JOIN (
    SELECT
        LTRIM(RTRIM(COST_CENTER)) AS CC,
        MAX(CC_DESCRIPTION) AS CC_DESCRIPTION
    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy
    GROUP BY LTRIM(RTRIM(COST_CENTER))
) h
    ON LTRIM(RTRIM(t.RCNTR)) = h.CC

WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
  AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

  AND LTRIM(RTRIM(t.RCNTR)) IN (
      'TDA87KE0',
      'TDB84KEM',
      'TDA87UK0',
      'TDB84UKD',
      'TDA87SR0',
      'TDB84SUR'
  )

GROUP BY LTRIM(RTRIM(t.RCNTR))

ORDER BY Cost_Center;
