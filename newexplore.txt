WITH SDS_Centers AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS CC,
        MAX(CC_DESCRIPTION) AS Cost_Center_Name,
        MAX(LEV04_DESC) AS Business_Unit
    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy
    WHERE LEV02 = 'NGRBT'
      AND LEV03_DESC = 'DS Strategic Deterrent Systems'
    GROUP BY
        UPPER(LTRIM(RTRIM(COST_CENTER)))
),

Selected_Costs AS (
    SELECT
        UPPER(LTRIM(RTRIM(t.RCNTR))) AS CC,

        COUNT(*) AS Transaction_Rows,

        SUM(
            TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ) AS Q1_Cost

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND TRY_CONVERT(BIGINT, t.RACCT) IN (
          4300591,
          4301050,
          4301049,
          4300625,
          4301102,
          4300703
      )

      AND EXISTS (
          SELECT 1
          FROM SDS_Centers h
          WHERE h.CC =
              UPPER(LTRIM(RTRIM(t.RCNTR)))
      )

    GROUP BY
        UPPER(LTRIM(RTRIM(t.RCNTR)))
)

SELECT TOP (25)
    s.CC AS Cost_Center,
    h.Cost_Center_Name,
    h.Business_Unit,
    s.Transaction_Rows,
    s.Q1_Cost

FROM Selected_Costs s

JOIN SDS_Centers h
    ON s.CC = h.CC

WHERE s.Q1_Cost <> 0

ORDER BY ABS(s.Q1_Cost) DESC;
