SET NOCOUNT ON;

;WITH FacilityAlias AS (
    SELECT *
    FROM (VALUES
        ('ONE AVIATION WAY',          '25167 ONE AVIATION WAY'),
        ('301 VOYAGER WAY NW',        '301 VOYAGER WAY NORTH WEST'),
        ('4065 COLONEL GLENN',        '4065 COLONEL GLENN HWY'),
        ('STRAWBERRY POINT',          '2828 STRAWBERRY POINT RD'),
        ('FAC WR PARK PLC',           '200 PARK PLACE DR'),
        ('FACILITY - WR IGNICO',      '98 IGNICO DR'),
        ('2300 DISCOVERY DR FL',      '2300 DISCOVERY DR'),
        ('8220 SAN PEDRO DR',         '8220 SAN PEDRO SUITE 300'),
        ('75 WEST ST WALPOLE',        '75 WEST ST')
    ) A (OLD_ADDRESS_KEY, NEW_ADDRESS_KEY)
),

Legacy AS (
    SELECT
        COALESCE(
            A.NEW_ADDRESS_KEY,
            X.ADDRESS_KEY
        ) AS FACILITY,

        SUM(X.COST) AS LEGACY_COST

    FROM (
        SELECT
            REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
                UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address])))),
                '.', ''),
                ',', ''),
                ' AVENUE', ' AVE'),
                ' ROAD', ' RD'),
                ' DRIVE', ' DR'),
                ' STREET', ' ST'),
                ' HIGHWAY', ' HWY'),
                ' PARKWAY', ' PKWY')
                AS ADDRESS_KEY,

            TRY_CONVERT(decimal(38,2), [Cost])
                AS COST

        FROM ecosystem_source.qmi.controllable_costs

        WHERE TRY_CONVERT(int, [Year]) = 2026
          AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))) = 'Q1'
          AND LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category]))) IN (
              '1 Labor',
              '2 Rent - Land & Building',
              '3 Depreciation & LHI',
              '4 Property Taxes',
              '5 Insurance',
              '6 Maintenance & Repairs',
              '7 Services',
              '8 Utilities'
          )
    ) X

    LEFT JOIN FacilityAlias A
      ON A.OLD_ADDRESS_KEY = X.ADDRESS_KEY

    GROUP BY
        COALESCE(A.NEW_ADDRESS_KEY, X.ADDRESS_KEY)
),

Hierarchy AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS COST_CENTER,

        ROW_NUMBER() OVER (
            PARTITION BY UPPER(LTRIM(RTRIM(COST_CENTER)))
            ORDER BY last_modified_date DESC, created_date DESC, id DESC
        ) AS RN

    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy

    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
      AND LTRIM(RTRIM(LEV03_DESC)) = 'DS C2 & Weapons Integration'
),

FacilityKey AS (
    SELECT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS COST_CENTER,

        MAX(NULLIF(LTRIM(RTRIM([Address])), ''))
            AS ADDRESS

    FROM ecosystem_source.qmi.costcenterkey

    GROUP BY
        UPPER(LTRIM(RTRIM([Cost Center])))
),

AdjustedNew AS (
    SELECT
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), K.ADDRESS)))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS FACILITY,

        SUM(
            TRY_CONVERT(decimal(38,2), T.KSL)
        ) AS NEW_COST

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 T

    JOIN Hierarchy H
      ON H.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))
     AND H.RN = 1

    JOIN FacilityKey K
      ON K.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))

    WHERE TRY_CONVERT(int, T.GJAHR) = 2026
      AND TRY_CONVERT(int, T.POPER) BETWEEN 1 AND 3
      AND TRY_CONVERT(decimal(38,2), T.KSL) IS NOT NULL

      AND (

          /* Existing labor population */
          (
              TRY_CONVERT(bigint, T.RACCT)
                  BETWEEN 4100000 AND 4109999
          )

          /* Existing list, with major NEW_ONLY accounts removed */
          OR TRY_CONVERT(bigint, T.RACCT) IN (

              8320037,

              /* Rent */
              4300591,
              -- 4300625 REMOVED
              4300887,
              4301077,

              /* Depreciation */
              4301048,
              -- 4301049 REMOVED
              4301050,
              4301231,

              /* Property Tax */
              4300626,

              /* Insurance */
              4300852,
              4300630,
              -- 8320479 REMOVED

              /* Maintenance */
              4300884,
              4300699,
              4300702,
              4300704,
              4300705,
              4300701,
              4300700,
              4300671,
              4300643,
              4300305,
              4300669,
              4300698,
              4300680,

              /* Services */
              4300703,
              -- 4300805 REMOVED
              4300666,
              4300807,

              /* Utilities */
              4301102,
              4301013,
              4301104,
              4300628,

              /* NEW: old-report GLs */
              4300652,
              4301101,
              4300395,
              4301103,
              7000301
          )
      )

    GROUP BY
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), K.ADDRESS)))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
),

Comparison AS (
    SELECT
        L.FACILITY,
        L.LEGACY_COST,
        N.NEW_COST,
        N.NEW_COST - L.LEGACY_COST
            AS DIFFERENCE

    FROM Legacy L

    JOIN AdjustedNew N
      ON N.FACILITY = L.FACILITY
)

SELECT
    COUNT(*) AS MATCHED_FACILITY_COUNT,

    SUM(LEGACY_COST)
        AS LEGACY_Q1_COST,

    SUM(NEW_COST)
        AS ADJUSTED_NEW_Q1_COST,

    SUM(DIFFERENCE)
        AS DIFFERENCE,

    CAST(
        100.0 * SUM(DIFFERENCE)
        / NULLIF(ABS(SUM(LEGACY_COST)), 0)

        AS decimal(12,2)
    ) AS DIFFERENCE_PCT

FROM Comparison;
