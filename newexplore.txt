DECLARE @CostCenter VARCHAR(100) = 'PUT_COST_CENTER_HERE';

SELECT
    TRY_CONVERT(INT, GJAHR) AS [Year],
    TRY_CONVERT(INT, POPER) AS [Month],

    COUNT(*) AS SAP_Rows,
    COUNT(DISTINCT RACCT) AS GL_Count,

    SUM(
        CASE WHEN TRY_CONVERT(DECIMAL(19,2), KSL) > 0
        THEN TRY_CONVERT(DECIMAL(19,2), KSL)
        ELSE 0 END
    ) AS Positive_Amount,

    SUM(
        CASE WHEN TRY_CONVERT(DECIMAL(19,2), KSL) < 0
        THEN TRY_CONVERT(DECIMAL(19,2), KSL)
        ELSE 0 END
    ) AS Negative_Amount,

    SUM(TRY_CONVERT(DECIMAL(19,2), KSL))
        AS Net_Amount,

    SUM(ABS(TRY_CONVERT(DECIMAL(19,2), KSL)))
        AS Gross_Activity

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03

WHERE LTRIM(RTRIM(RCNTR)) = @CostCenter

GROUP BY
    TRY_CONVERT(INT, GJAHR),
    TRY_CONVERT(INT, POPER)

ORDER BY [Year], [Month];
