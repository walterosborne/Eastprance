SELECT
    CASE
        WHEN GROUPING(x.Cost_Category) = 1
            THEN 'TOTAL'
        ELSE x.Cost_Category
    END AS Cost_Category,

    SUM(CASE
        WHEN x.Source = 'Priscilla'
            THEN x.Amount
        ELSE 0
    END) AS Priscilla_Q1,

    SUM(CASE
        WHEN x.Source = 'SAP'
            THEN x.Amount
        ELSE 0
    END) AS SAP_CWI_Q1,

    SUM(CASE
        WHEN x.Source = 'SAP'
            THEN x.Amount
        ELSE -x.Amount
    END) AS SAP_Minus_Priscilla,

    SUM(CASE
        WHEN x.Source = 'SAP'
            THEN x.Review_Amount
        ELSE 0
    END) AS SAP_Review_Only

FROM (

    /* Priscilla: populated Cost Element = CWI proxy */
    SELECT
        'Priscilla' AS Source,

        LTRIM(RTRIM(c.[Cost Category]))
            AS Cost_Category,

        TRY_CONVERT(
            DECIMAL(19,2),
            c.[Cost]
        ) AS Amount,

        CAST(0 AS DECIMAL(19,2))
            AS Review_Amount

    FROM ecosystem_source.qmi.controllable_costs c

    WHERE TRY_CONVERT(INT, c.[Year]) = 2026

      AND LTRIM(RTRIM(c.[Quarter])) = 'Q1'

      AND c.[Cost Element] IS NOT NULL


    UNION ALL


    /* SAP: mapped CWI facilities */
    SELECT
        'SAP' AS Source,

        sr.Cost_Category,

        CASE
            WHEN sr.Mapping_Status = 'Selected'
                THEN sr.Amount
            ELSE 0
        END AS Amount,

        CASE
            WHEN sr.Mapping_Status = 'Review'
                THEN sr.Amount
            ELSE 0
        END AS Review_Amount

    FROM SelectedRows sr

    WHERE sr.Division =
        'DS C2 & Weapons Integration'

      AND sr.Reporting_Scope =
        'Physical facility'

) x

GROUP BY ROLLUP(x.Cost_Category)

ORDER BY
    GROUPING(x.Cost_Category),
    x.Cost_Category;
