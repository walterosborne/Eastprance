-- QMI Customer Escapes: next discovery queries
-- Run this entire file in SQL Server Management Studio.
-- Read-only; returns four result grids.
USE [Quality Data Mart];

-- 1. Columns in the two candidate monthly views.
SELECT
    s.name AS schema_name,
    o.name AS view_name,
    c.column_id,
    c.name AS column_name
FROM sys.columns AS c
JOIN sys.objects AS o ON o.object_id = c.object_id
JOIN sys.schemas AS s ON s.schema_id = o.schema_id
WHERE (s.name = 'QAF'
       AND o.name = 'Monthly Data (metrics and data)')
   OR (s.name = 'Grand Central'
       AND o.name = 'Monthly Metrics - underlying data')
ORDER BY s.name, o.name, c.column_id;

-- 2. Find the Delivered Quality metrics and their component IDs.
-- Keep candidate metrics for review before choosing which values to sum.
SELECT
    [METC-DATV_ID],
    [Metric Level],
    [Metric],
    [Metric Title (indented)],
    [Parent Metric (with level)],
    [Metric Status]
FROM [QAF].[QAF Metrics]
WHERE [Metric] LIKE '%deliver%'
   OR [Metric] LIKE '%defect%'
   OR [Metric] LIKE '%escap%'
   OR [Metric Title (indented)] LIKE '%deliver%'
   OR [Metric Title (indented)] LIKE '%defect%'
   OR [Metric Title (indented)] LIKE '%escap%'
   OR [Parent Metric (with level)] LIKE '%Delivered Quality%'
ORDER BY [Metric Sort Order];

-- 3. Check whether the historical raw schema is visible.
SELECT name AS schema_name
FROM sys.schemas
WHERE name LIKE '%QAF%'
   OR name LIKE '%Raw%'
ORDER BY name;

-- 4. Check visibility of the raw monthly history/configuration/version tables.
-- NULL can mean the object is absent or metadata permissions hide it.
SELECT
    OBJECT_ID(N'[QAF_RawData].[PMDH]') AS PMDH_object_id,
    OBJECT_ID(N'[QAF_RawData].[PCMV]') AS PCMV_object_id,
    OBJECT_ID(N'[QAF_RawData].[MCVR]') AS MCVR_object_id;
