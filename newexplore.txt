-- Top roster sites by absolute hours, with labor aggregated before the roster join.
WITH labor_by_person AS (
    SELECT
        UPPER(LTRIM(RTRIM(a.MYID))) AS my_id,
        SUM(ABS(hours.quantity)) AS absolute_hours,
        COUNT_BIG(*) AS labor_rows
    FROM [DTO_Business_Management].[rpt].[rb_Actuals_RM_Load_Table] AS a
    CROSS APPLY (VALUES (
        TRY_CONVERT(int, RIGHT(LTRIM(RTRIM(a.[Period])), 4)),
        TRY_CONVERT(int, LEFT(LTRIM(RTRIM(a.[Period])), 2))
    )) AS period(labor_year, labor_month)
    CROSS APPLY (VALUES (TRY_CONVERT(decimal(18, 2), a.Hours))) AS hours(quantity)
    WHERE NULLIF(LTRIM(RTRIM(a.MYID)), '') IS NOT NULL
      AND NULLIF(LTRIM(RTRIM(a.Cost_Center)), '') IS NOT NULL
      AND period.labor_year >= 2025
      AND period.labor_month BETWEEN 1 AND 12
      AND period.labor_year * 100 + period.labor_month < YEAR(GETUTCDATE()) * 100 + MONTH(GETUTCDATE())
      AND hours.quantity IS NOT NULL
      AND EXISTS (
          SELECT 1
          FROM [DTO_Business_Management].[rpt].[rb_load_cost_center_hierarchy] AS h
          WHERE LTRIM(RTRIM(h.LEV02)) = 'NGRBT'
            AND UPPER(LTRIM(RTRIM(h.COST_CENTER))) = UPPER(LTRIM(RTRIM(a.Cost_Center)))
      )
    GROUP BY UPPER(LTRIM(RTRIM(a.MYID)))
),
roster_facilities AS (
    SELECT
        UPPER(LTRIM(RTRIM(ids.my_id))) AS my_id,
        CONCAT_WS(', ', LTRIM(RTRIM(r.Work_City_NSR)),
            NULLIF(LTRIM(RTRIM(r.Work_State_NSR)), '')) AS facility
    FROM [DTO_Business_Management].[dbo].[rb_roster_transform] AS r
    CROSS APPLY (VALUES
        (r.Employee_MyID_NSR),
        (r.Employee_MyID_SR),
        (r.Employee_MyID_DM),
        (r.Employee_MyID_SAP)
    ) AS ids(my_id)
    WHERE NULLIF(LTRIM(RTRIM(ids.my_id)), '') IS NOT NULL
      AND NULLIF(LTRIM(RTRIM(r.Work_City_NSR)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(r.Work_City_NSR))) NOT IN ('UNKNOWN CITY', 'UNKNOWN', 'N/A')
),
mapped_people AS (
    SELECT my_id, MIN(facility) AS facility
    FROM roster_facilities
    GROUP BY my_id
    HAVING MIN(facility) = MAX(facility)
)
SELECT TOP (30)
    p.facility,
    SUM(a.absolute_hours) AS absolute_hours,
    SUM(a.labor_rows) AS labor_rows,
    COUNT_BIG(*) AS employees
FROM labor_by_person AS a
INNER JOIN mapped_people AS p ON p.my_id = a.my_id
GROUP BY p.facility
ORDER BY absolute_hours DESC, p.facility;
