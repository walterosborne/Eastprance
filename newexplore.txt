SELECT
    COLUMN_NAME,
    DATA_TYPE
FROM DTO_Business_Management.INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_SCHEMA = 'src'
  AND TABLE_NAME = 'rb_CVG_Transaction_Details_03'
  AND (
       COLUMN_NAME LIKE '%SL%'
    OR COLUMN_NAME LIKE '%AMT%'
    OR COLUMN_NAME LIKE '%AMOUNT%'
    OR COLUMN_NAME LIKE '%DOC%'
    OR COLUMN_NAME LIKE '%BELNR%'
    OR COLUMN_NAME LIKE '%BUDAT%'
    OR COLUMN_NAME LIKE '%BLART%'
    OR COLUMN_NAME LIKE '%SGTXT%'
    OR COLUMN_NAME LIKE '%BKTXT%'
  )
ORDER BY ORDINAL_POSITION;
