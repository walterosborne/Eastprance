SELECT
    "ACCT_ID",
    "ACCT_NAME",
    "L1_ACCT_SEG_ID",
    "L1_ACCT_NAME",
    "L2_ACCT_SEG_ID",
    "L2_ACCT_NAME",
    "L3_ACCT_SEG_ID",
    "L3_ACCT_NAME",
    "L4_ACCT_SEG_ID",
    "L4_ACCT_NAME",
    "L5_ACCT_SEG_ID",
    "L5_ACCT_NAME",
    "L6_ACCT_SEG_ID",
    "L6_ACCT_NAME",
    "L7_ACCT_SEG_ID",
    "L7_ACCT_NAME",
    COUNT(*) AS "ROW_COUNT",
    SUM("AMT") AS "AMT"

FROM "_SYS_BIC"."ngc.DWH.SectorContent.NGSS.SP_COSTPOINT.Restricted/CVG_CP_GL_DETL_RESTRICTED"
(
    'PLACEHOLDER' = ('$$IP_FY_PD_NO_TO$$', '202608'),
    'PLACEHOLDER' = ('$$IP_ACCT_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_ORG_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_L1_PROJ_SEG_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_FY_PD_NO_FROM$$', '202601'),
    'PLACEHOLDER' = ('$$IP_INCL_ITD_PD_NO$$', 'Y'),
    'PLACEHOLDER' = ('$$IP_INCL_ALL_HDR$$', 'N')
)

WHERE "CP_INSTANCE" = 'CPENT'
  AND "L1_ORG_NAME" = 'MSG'
  AND "L3_ORG_NAME" IN (
      'ARMAMENT SYSTEMS',
      'Advanced Weapons',
      'ROCKET CENTER (ABL)',
      'ELKTON',
      'NRE',
      'PROPULSION SYSTEMS',
      'WS TPS'
  )

GROUP BY
    "ACCT_ID",
    "ACCT_NAME",
    "L1_ACCT_SEG_ID",
    "L1_ACCT_NAME",
    "L2_ACCT_SEG_ID",
    "L2_ACCT_NAME",
    "L3_ACCT_SEG_ID",
    "L3_ACCT_NAME",
    "L4_ACCT_SEG_ID",
    "L4_ACCT_NAME",
    "L5_ACCT_SEG_ID",
    "L5_ACCT_NAME",
    "L6_ACCT_SEG_ID",
    "L6_ACCT_NAME",
    "L7_ACCT_SEG_ID",
    "L7_ACCT_NAME"

ORDER BY "AMT" DESC;
