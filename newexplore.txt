SELECT
    [Segment],

    COUNT(*) AS Row_Count,

    SUM(TRY_CONVERT(DECIMAL(19,2), [Cost]))
        AS Total_Cost,

    SUM(CASE
        WHEN NULLIF(
            LTRIM(RTRIM(CONVERT(VARCHAR(100), [Cost Element]))),
            ''
        ) IS NOT NULL
        THEN TRY_CONVERT(DECIMAL(19,2), [Cost])
        ELSE 0
    END) AS Cost_With_Element,

    SUM(CASE
        WHEN NULLIF(
            LTRIM(RTRIM(CONVERT(VARCHAR(100), [Cost Element]))),
            ''
        ) IS NULL
        THEN TRY_CONVERT(DECIMAL(19,2), [Cost])
        ELSE 0
    END) AS Cost_Without_Element

FROM ecosystem_source.qmi.controllable_costs

WHERE [Quarter] = 'Q1'
  AND [Year] = 2026

GROUP BY [Segment]

ORDER BY Total_Cost DESC;
