WITH KeyCenters AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS CC
    FROM ecosystem_source.qmi.costcenterkey
),

ERP AS (
    SELECT
        TRY_CONVERT(
            INT,
            LEFT(e.[Fiscal Year Period], 3)
        ) AS Month,

        UPPER(LTRIM(RTRIM(e.[Cost Center]))) AS CC,

        COUNT(*) AS ERP_Rows,

        SUM(
            TRY_CONVERT(DECIMAL(19,2), e.[Actual Amount])
        ) AS ERP_Net,

        SUM(
            ABS(TRY_CONVERT(
                DECIMAL(19,2), e.[Actual Amount]
            ))
        ) AS ERP_Gross

    FROM ecosystem_source.dbo.erpdata e

    WHERE e.[Fiscal Year Period] IN (
        '001/2026', '002/2026', '003/2026'
    )

    AND EXISTS (
        SELECT 1
        FROM KeyCenters k
        WHERE k.CC =
            UPPER(LTRIM(RTRIM(e.[Cost Center])))
    )

    GROUP BY
        TRY_CONVERT(
            INT,
            LEFT(e.[Fiscal Year Period], 3)
        ),
        UPPER(LTRIM(RTRIM(e.[Cost Center])))
),

SAP AS (
    SELECT
        TRY_CONVERT(INT, s.POPER) AS Month,

        UPPER(LTRIM(RTRIM(s.RCNTR))) AS CC,

        COUNT(*) AS SAP_Rows,

        SUM(
            TRY_CONVERT(DECIMAL(19,2), s.KSL)
        ) AS SAP_Net,

        SUM(
            ABS(TRY_CONVERT(DECIMAL(19,2), s.KSL))
        ) AS SAP_Gross

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 s

    WHERE TRY_CONVERT(INT, s.GJAHR) = 2026
      AND TRY_CONVERT(INT, s.POPER) BETWEEN 1 AND 3

      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE k.CC =
              UPPER(LTRIM(RTRIM(s.RCNTR)))
      )

    GROUP BY
        TRY_CONVERT(INT, s.POPER),
        UPPER(LTRIM(RTRIM(s.RCNTR)))
)

SELECT TOP (30)
    COALESCE(e.Month, s.Month) AS Month,
    COALESCE(e.CC, s.CC) AS Cost_Center,

    e.ERP_Rows,
    s.SAP_Rows,

    e.ERP_Net,
    s.SAP_Net,

    e.ERP_Gross,
    s.SAP_Gross,

    COALESCE(e.ERP_Gross, 0)
      - COALESCE(s.SAP_Gross, 0)
        AS Gross_Difference

FROM ERP e

FULL OUTER JOIN SAP s
    ON e.Month = s.Month
   AND e.CC = s.CC

ORDER BY
    ABS(
        COALESCE(e.ERP_Gross, 0)
        - COALESCE(s.SAP_Gross, 0)
    ) DESC;
