SELECT
    ACCT_LEVEL01,
    ACCT_LEVEL01_TEXT,
    ACCT_LEVEL02,
    ACCT_LEVEL02_TEXT,
    ACCT_LEVEL03,
    ACCT_LEVEL03_TEXT,
    ACCT_LEVEL04,
    ACCT_LEVEL04_TEXT,
    COUNT(*) AS row_count,
    SUM(TRY_CONVERT(DECIMAL(18,2), KSL)) AS cost
FROM src.rb_CVG_Transaction_Details_03
WHERE
    TRY_CONVERT(INT, GJAHR) IN (2025, 2026)
    AND TRY_CONVERT(INT, POPER) BETWEEN 1 AND 12
GROUP BY
    ACCT_LEVEL01,
    ACCT_LEVEL01_TEXT,
    ACCT_LEVEL02,
    ACCT_LEVEL02_TEXT,
    ACCT_LEVEL03,
    ACCT_LEVEL03_TEXT,
    ACCT_LEVEL04,
    ACCT_LEVEL04_TEXT
ORDER BY
    ABS(SUM(TRY_CONVERT(DECIMAL(18,2), KSL))) DESC;
