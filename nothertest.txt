SELECT
    ACCT_LEVEL01_TEXT,
    ACCT_LEVEL02_TEXT,
    COUNT(*) AS row_count,
    SUM(TRY_CONVERT(DECIMAL(18,2), KSL)) AS cost,
    MIN(TRY_CONVERT(INT, POPER)) AS first_month,
    MAX(TRY_CONVERT(INT, POPER)) AS latest_month
FROM src.rb_CVG_Transaction_Details_03
WHERE
    TRY_CONVERT(INT, GJAHR) = 2026
    AND TRY_CONVERT(INT, POPER) BETWEEN 1 AND 12
GROUP BY
    ACCT_LEVEL01_TEXT,
    ACCT_LEVEL02_TEXT
ORDER BY
    ABS(SUM(TRY_CONVERT(DECIMAL(18,2), KSL))) DESC;
