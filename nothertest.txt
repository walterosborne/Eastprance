SELECT
    ACCT_LEVEL03_TEXT,
    ACCT_LEVEL04_TEXT,
    COUNT(*) AS row_count,
    SUM(TRY_CONVERT(DECIMAL(18,2), KSL)) AS cost
FROM src.rb_CVG_Transaction_Details_03
WHERE
    GJAHR = '2026'
    AND POPER = '008'
    AND ACCT_LEVEL02_TEXT = 'NGRB Indirect Non Labor CEG'
GROUP BY
    ACCT_LEVEL03_TEXT,
    ACCT_LEVEL04_TEXT
ORDER BY
    ABS(SUM(TRY_CONVERT(DECIMAL(18,2), KSL))) DESC;
