WITH ERP AS (
    SELECT
        TRY_CONVERT(BIGINT, [G/L Account]) AS GL,
        COUNT(*) AS ERP_Rows,
        SUM(TRY_CONVERT(DECIMAL(19,2), [Actual Amount]))
            AS ERP_Net,
        SUM(ABS(TRY_CONVERT(DECIMAL(19,2), [Actual Amount])))
            AS ERP_Gross
    FROM ecosystem_source.dbo.erpdata
    GROUP BY TRY_CONVERT(BIGINT, [G/L Account])
),
SAP AS (
    SELECT DISTINCT
        TRY_CONVERT(BIGINT, RACCT) AS GL
    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03
    WHERE TRY_CONVERT(INT, GJAHR) = 2026
      AND TRY_CONVERT(INT, POPER) BETWEEN 6 AND 8
)
SELECT
    CASE
        WHEN SAP.GL IS NULL THEN 'Missing from SAP'
        ELSE 'Found in SAP'
    END AS Match_Status,

    COUNT(*) AS GL_Accounts,
    SUM(ERP.ERP_Rows) AS ERP_Rows,
    SUM(ERP.ERP_Net) AS ERP_Net,
    SUM(ERP.ERP_Gross) AS ERP_Gross

FROM ERP
LEFT JOIN SAP
    ON ERP.GL = SAP.GL

GROUP BY
    CASE
        WHEN SAP.GL IS NULL THEN 'Missing from SAP'
        ELSE 'Found in SAP'
    END;
