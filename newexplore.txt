WITH LegacyQ1 AS (
    SELECT
        TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(
                DECIMAL(38,6),
                REPLACE(
                    LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                    ',',
                    ''
                )
            )
        ) AS GL,
        TRY_CONVERT(DECIMAL(38,2), [Cost]) AS COST
    FROM ecosystem_source.qmi.controllable_costs
    WHERE TRY_CONVERT(INT, [Year]) = 2026
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))) = 'Q1'
),
LegacySelected AS (
    SELECT COST
    FROM LegacyQ1
    WHERE
           GL BETWEEN 4100000 AND 4109999
        OR GL IN (
            8320037,
            4300591, 4300625, 4300887, 4301077,
            4301048, 4301049, 4301050, 4301231,
            4300626,
            4300852, 4300630, 8320479,
            4300884, 4300699, 4300702, 4300704,
            4300705, 4300701, 4300700, 4300671,
            4300643, 4300305, 4300669, 4300698, 4300680,
            4300703, 4300805, 4300666, 4300807,
            4301102, 4301013, 4301104, 4300628
        )
),
NewCWI AS (
    SELECT TRY_CONVERT(DECIMAL(38,2), cost) AS COST
    FROM ecosystem_source.qmi.controllable_costs_new_extract
    WHERE [year] = 2026
      AND [month] BETWEEN 1 AND 3
      AND division = 'DS C2 & Weapons Integration'
)
SELECT
    (SELECT SUM(COST) FROM LegacyQ1)
        AS LEGACY_ALL_Q1_COST,

    (SELECT SUM(COST)
     FROM LegacyQ1
     WHERE GL IS NOT NULL)
        AS LEGACY_Q1_WITH_GL_COST,

    (SELECT SUM(COST) FROM LegacySelected)
        AS LEGACY_SELECTED_GL_COST,

    (SELECT SUM(COST) FROM NewCWI)
        AS NEW_CWI_SELECTED_GL_COST,

    (SELECT SUM(COST) FROM NewCWI)
      - (SELECT SUM(COST) FROM LegacySelected)
        AS DIFFERENCE,

    CAST(
        100.0 *
        (
            (SELECT SUM(COST) FROM NewCWI)
          - (SELECT SUM(COST) FROM LegacySelected)
        )
        / NULLIF(
            ABS((SELECT SUM(COST) FROM LegacySelected)),
            0
        )
        AS DECIMAL(12,2)
    ) AS DIFFERENCE_PCT;
