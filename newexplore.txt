-- 1. Inspect G/L 8999999, including its description if present.
-- It was ERP-only in our January comparison, so SAP's
-- GL_TXT20 cannot supply its description there.

SELECT TOP (10)
    e.*
FROM ecosystem_source.dbo.erpdata AS e
WHERE TRY_CONVERT(BIGINT, e.[G/L Account]) = 8999999
  AND e.[Fiscal Year Period] IN (
      '001/2026', '002/2026', '003/2026'
  )
ORDER BY
    e.[Fiscal Year Period],
    e.[Cost Center];


-- 2. Find the remaining gross discrepancies by G/L account.

WITH KeyCenters AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS CC
    FROM ecosystem_source.qmi.costcenterkey
),

ERP AS (
    SELECT
        TRY_CONVERT(BIGINT, e.[G/L Account]) AS GL,

        COUNT(*) AS ERP_Rows,

        SUM(
            TRY_CONVERT(DECIMAL(19,2), e.[Actual Amount])
        ) AS ERP_Net,

        SUM(
            ABS(TRY_CONVERT(
                DECIMAL(19,2), e.[Actual Amount]
            ))
        ) AS ERP_Gross

    FROM ecosystem_source.dbo.erpdata AS e

    WHERE e.[Fiscal Year Period] IN (
        '001/2026', '002/2026', '003/2026'
    )

      AND TRY_CONVERT(BIGINT, e.[G/L Account])
          <> 8999999

      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE k.CC =
              UPPER(LTRIM(RTRIM(e.[Cost Center])))
      )

    GROUP BY
        TRY_CONVERT(BIGINT, e.[G/L Account])
),

SAP AS (
    SELECT
        TRY_CONVERT(BIGINT, s.RACCT) AS GL,

        MAX(NULLIF(
            LTRIM(RTRIM(s.GL_TXT20)), ''
        )) AS SAP_Account_Description,

        COUNT(*) AS SAP_Rows,

        SUM(
            TRY_CONVERT(DECIMAL(19,2), s.KSL)
        ) AS SAP_Net,

        SUM(
            ABS(TRY_CONVERT(DECIMAL(19,2), s.KSL))
        ) AS SAP_Gross

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 AS s

    WHERE TRY_CONVERT(INT, s.GJAHR) = 2026
      AND TRY_CONVERT(INT, s.POPER) BETWEEN 1 AND 3

      AND TRY_CONVERT(BIGINT, s.RACCT)
          <> 8999999

      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE k.CC =
              UPPER(LTRIM(RTRIM(s.RCNTR)))
      )

    GROUP BY
        TRY_CONVERT(BIGINT, s.RACCT)
)

SELECT TOP (30)
    COALESCE(e.GL, s.GL) AS GL,

    s.SAP_Account_Description,

    e.ERP_Rows,
    s.SAP_Rows,

    e.ERP_Net,
    s.SAP_Net,

    e.ERP_Gross,
    s.SAP_Gross,

    COALESCE(e.ERP_Gross, 0)
      - COALESCE(s.SAP_Gross, 0)
        AS Gross_Difference,

    CASE
        WHEN e.GL IS NULL THEN 'SAP only'
        WHEN s.GL IS NULL THEN 'ERP only'
        ELSE 'Both'
    END AS Match_Status

FROM ERP AS e

FULL OUTER JOIN SAP AS s
    ON e.GL = s.GL

ORDER BY
    ABS(
        COALESCE(e.ERP_Gross, 0)
        - COALESCE(s.SAP_Gross, 0)
    ) DESC;
