SELECT
    [Fiscal Year Period],
    COUNT(*) AS ERP_Rows,
    COUNT(DISTINCT [Cost Center]) AS Cost_Centers,
    COUNT(DISTINCT [G/L Account]) AS GL_Accounts,
    SUM(TRY_CONVERT(DECIMAL(19,2), [Actual Amount]))
        AS Net_Amount,
    SUM(ABS(TRY_CONVERT(DECIMAL(19,2), [Actual Amount])))
        AS Gross_Activity
FROM ecosystem_source.dbo.erpdata
GROUP BY [Fiscal Year Period]
ORDER BY
    TRY_CONVERT(INT, RIGHT([Fiscal Year Period], 4)),
    TRY_CONVERT(INT, LEFT([Fiscal Year Period], 3));
