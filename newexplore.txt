-- 1. Priscilla's central SDS Other records
SELECT *
FROM ecosystem_source.qmi.controllable_costs
WHERE TRY_CONVERT(INT, [Year]) = 2026
  AND LTRIM(RTRIM([Quarter])) = 'Q1'
  AND [Address] LIKE '%Strategic Deterrent%'
  AND LTRIM(RTRIM([Cost Category])) = '0 Other';


-- 2. SAP's $4.08M overhead depreciation
SELECT
    t.RCNTR AS Cost_Center,
    t.GJAHR AS Fiscal_Year,
    t.POPER AS Fiscal_Period,
    t.RACCT AS GL_Account,
    t.GL_TXT20 AS Description,
    TRY_CONVERT(DECIMAL(19,2), t.KSL) AS Amount
FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t
WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
  AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3
  AND LTRIM(RTRIM(t.RCNTR)) = 'TDA10H1C'
  AND TRY_CONVERT(BIGINT, t.RACCT)
      IN (4301049, 4301050)
ORDER BY
    TRY_CONVERT(INT, t.POPER),
    t.RACCT;
