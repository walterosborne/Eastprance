/* ============================================================
   CWI Q1 2026 old-vs-new facility reconciliation
   NO TEMP TABLES.

   OLD:
   - Full legacy facility-report categories
   - No Cost Element requirement

   NEW:
   - DS C2 & Weapons Integration
   - Already mapped RCNTR -> qmi.costcenterkey
   - Already filtered to selected facility G/L accounts

   Includes manual aliases for obvious legacy/new facility naming
   differences discovered in the previous run.
   ============================================================ */

SET NOCOUNT ON;


/* ============================================================
   RESULT 1 — Overall comparison for facilities present on BOTH
   sides after applying known aliases.
   ============================================================ */

;WITH FacilityAlias AS (
    SELECT *
    FROM (VALUES
        ('ONE AVIATION WAY',          '25167 ONE AVIATION WAY'),
        ('301 VOYAGER WAY NW',        '301 VOYAGER WAY NORTH WEST'),
        ('4065 COLONEL GLENN',        '4065 COLONEL GLENN HWY'),
        ('STRAWBERRY POINT',          '2828 STRAWBERRY POINT RD'),
        ('FAC WR PARK PLC',           '200 PARK PLACE DR'),
        ('FACILITY - WR IGNICO',      '98 IGNICO DR'),
        ('2300 DISCOVERY DR FL',      '2300 DISCOVERY DR'),
        ('8220 SAN PEDRO DR',         '8220 SAN PEDRO SUITE 300'),
        ('75 WEST ST WALPOLE',        '75 WEST ST')
    ) A (OLD_ADDRESS_KEY, NEW_ADDRESS_KEY)
),

LegacyRaw AS (
    SELECT
        LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address]))) AS ADDRESS_DISPLAY,
        LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category]))) AS COST_CATEGORY,
        TRY_CONVERT(decimal(38,2), [Cost]) AS COST
    FROM ecosystem_source.qmi.controllable_costs
    WHERE TRY_CONVERT(int, [Year]) = 2026
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))) = 'Q1'
      AND LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category]))) IN (
          '1 Labor',
          '2 Rent - Land & Building',
          '3 Depreciation & LHI',
          '4 Property Taxes',
          '5 Insurance',
          '6 Maintenance & Repairs',
          '7 Services',
          '8 Utilities'
      )
),

LegacyNormalized AS (
    SELECT
        ADDRESS_DISPLAY,
        COST_CATEGORY,
        COST,

        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(ADDRESS_DISPLAY),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS ADDRESS_KEY

    FROM LegacyRaw
    WHERE NULLIF(ADDRESS_DISPLAY, '') IS NOT NULL
),

LegacyBridged AS (
    SELECT
        COALESCE(
            A.NEW_ADDRESS_KEY,
            L.ADDRESS_KEY
        ) AS ADDRESS_KEY,

        L.COST_CATEGORY,

        SUM(L.COST) AS LEGACY_COST

    FROM LegacyNormalized L

    LEFT JOIN FacilityAlias A
      ON A.OLD_ADDRESS_KEY = L.ADDRESS_KEY

    GROUP BY
        COALESCE(
            A.NEW_ADDRESS_KEY,
            L.ADDRESS_KEY
        ),
        L.COST_CATEGORY
),

LegacyFacility AS (
    SELECT
        ADDRESS_KEY,
        SUM(LEGACY_COST) AS LEGACY_COST
    FROM LegacyBridged
    GROUP BY ADDRESS_KEY
),

NewRaw AS (
    SELECT
        LTRIM(RTRIM(CONVERT(nvarchar(1000), facility))) AS ADDRESS_DISPLAY,
        LTRIM(RTRIM(CONVERT(nvarchar(255), cost_category))) AS COST_CATEGORY,
        TRY_CONVERT(decimal(38,2), cost) AS COST
    FROM ecosystem_source.qmi.controllable_costs_new_extract
    WHERE [year] = 2026
      AND [month] BETWEEN 1 AND 3
      AND LTRIM(RTRIM(division)) = 'DS C2 & Weapons Integration'
),

NewNormalized AS (
    SELECT
        ADDRESS_DISPLAY,
        COST_CATEGORY,
        COST,

        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(ADDRESS_DISPLAY),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS ADDRESS_KEY

    FROM NewRaw
    WHERE NULLIF(ADDRESS_DISPLAY, '') IS NOT NULL
),

NewFacility AS (
    SELECT
        ADDRESS_KEY,
        SUM(COST) AS NEW_COST
    FROM NewNormalized
    GROUP BY ADDRESS_KEY
),

CommonFacility AS (
    SELECT
        L.ADDRESS_KEY,
        L.LEGACY_COST,
        N.NEW_COST
    FROM LegacyFacility L
    JOIN NewFacility N
      ON N.ADDRESS_KEY = L.ADDRESS_KEY
)

SELECT
    COUNT(*) AS COMMON_FACILITY_COUNT,

    SUM(LEGACY_COST)
        AS LEGACY_COMMON_Q1_COST,

    SUM(NEW_COST)
        AS NEW_COMMON_Q1_COST,

    SUM(NEW_COST) - SUM(LEGACY_COST)
        AS DIFFERENCE,

    CAST(
        100.0 *
        (
            SUM(NEW_COST) - SUM(LEGACY_COST)
        )
        / NULLIF(ABS(SUM(LEGACY_COST)), 0)

        AS decimal(12,2)
    ) AS DIFFERENCE_PCT

FROM CommonFacility;

        AS LEGACY_COST,

    COALESCE(N.NEW_COST, 0)
        AS NEW_COST,

    COALESCE(N.NEW_COST, 0)
      - COALESCE(L.LEGACY_COST, 0)
        AS DIFFERENCE,

    CAST(
        100.0 *
        (
            COALESCE(N.NEW_COST, 0)

        )
        / NULLIF(
            ABS(COALESCE(L.LEGACY_COST, 0)),
            0
        )

        AS decimal(12,2)
    ) AS DIFFERENCE_PCT

FROM LegacyCategory L

FULL OUTER JOIN NewCategory N
  ON N.COST_CATEGORY = L.COST_CATEGORY

ORDER BY COST_CATEGORY;
