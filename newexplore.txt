WITH SDS_Expenses AS (
    SELECT
        UPPER(LTRIM(RTRIM(t.RCNTR))) AS Cost_Center,
        TRY_CONVERT(BIGINT, t.RACCT) AS GL,
        TRY_CONVERT(DECIMAL(19,2), t.KSL) AS Amount

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND TRY_CONVERT(BIGINT, t.RACCT) IN (
          4300591,  -- Real estate lease
          4301050,  -- LHI amortization
          4301049,  -- Machinery depreciation
          4300625,  -- Variable rent / CAM
          4301102,  -- Purchased electricity
          4300703   -- Janitorial
      )

      AND EXISTS (
          SELECT 1
          FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy h

          WHERE UPPER(LTRIM(RTRIM(h.COST_CENTER)))
              = UPPER(LTRIM(RTRIM(t.RCNTR)))

            AND h.LEV02 = 'NGRBT'
            AND h.LEV03_DESC =
                'DS Strategic Deterrent Systems'
      )
)

SELECT TOP (30)
    Cost_Center,

    COUNT(*) AS Transaction_Rows,

    SUM(CASE WHEN GL = 4300591
        THEN Amount ELSE 0 END) AS Real_Estate_Lease,

    SUM(CASE WHEN GL = 4301050
        THEN Amount ELSE 0 END) AS LHI_Amortization,

    SUM(CASE WHEN GL = 4301049
        THEN Amount ELSE 0 END) AS Machinery_Depreciation,

    SUM(CASE WHEN GL = 4300625
        THEN Amount ELSE 0 END) AS Variable_Rent_CAM,

    SUM(CASE WHEN GL = 4301102
        THEN Amount ELSE 0 END) AS Electricity,

    SUM(CASE WHEN GL = 4300703
        THEN Amount ELSE 0 END) AS Janitorial,

    SUM(Amount) AS Total_Selected_Cost,

    SUM(SUM(Amount)) OVER ()
        AS All_SDS_Selected_Cost

FROM SDS_Expenses

GROUP BY Cost_Center

ORDER BY Total_Selected_Cost DESC;
