SELECT
    [Address],
    [Cost Category],
    COUNT(*) AS Report_Rows,

    SUM(
        TRY_CONVERT(DECIMAL(19,2), [Cost])
    ) AS Q1_Cost,

    SUM(
        SUM(TRY_CONVERT(DECIMAL(19,2), [Cost]))
    ) OVER () AS Total_Central_SDS

FROM ecosystem_source.qmi.controllable_costs

WHERE TRY_CONVERT(INT, [Year]) = 2026
  AND LTRIM(RTRIM([Quarter])) = 'Q1'

  AND UPPER([Address])
      LIKE '%STRATEGIC DETERRENT%'

GROUP BY
    [Address],
    [Cost Category]

ORDER BY
    [Address],
    [Cost Category];
