SELECT
    [Fiscal Year],
    [Period],

    COUNT(*) AS Row_Count,

    COUNT(DISTINCT [Cost Center]) AS Cost_Center_Count,

    COUNT(DISTINCT [G/L Account]) AS GL_Account_Count,

    SUM(TRY_CONVERT(DECIMAL(19, 2), [Actual Amount]))
        AS Net_Amount,

    SUM(
        CASE
            WHEN TRY_CONVERT(DECIMAL(19, 2), [Actual Amount]) > 0
            THEN TRY_CONVERT(DECIMAL(19, 2), [Actual Amount])
            ELSE 0
        END
    ) AS Positive_Amount,

    SUM(
        CASE
            WHEN TRY_CONVERT(DECIMAL(19, 2), [Actual Amount]) < 0
            THEN TRY_CONVERT(DECIMAL(19, 2), [Actual Amount])
            ELSE 0
        END
    ) AS Negative_Amount

FROM ecosystem_source.dbo.erpdata

GROUP BY
    [Fiscal Year],
    [Period]

ORDER BY
    [Fiscal Year],
    [Period];
