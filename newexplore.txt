WITH OtherCosts AS (
    SELECT
        TRY_CONVERT(BIGINT, t.RACCT) AS GL,
        t.GL_TXT20 AS Description,
        LTRIM(RTRIM(t.RCNTR)) AS Cost_Center,
        TRY_CONVERT(DECIMAL(19,2), t.KSL) AS Amount

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND TRY_CONVERT(BIGINT, t.RACCT) NOT IN (
          4300591, 4300625,
          4301050, 4301049, 4301048,
          4301102, 4300703
      )

      AND EXISTS (
          SELECT 1
          FROM ecosystem_source.qmi.costcenterkey k
          WHERE UPPER(LTRIM(RTRIM(k.[Cost Center])))
              = UPPER(LTRIM(RTRIM(t.RCNTR)))
      )

      AND EXISTS (
          SELECT 1
          FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy h
          WHERE UPPER(LTRIM(RTRIM(h.COST_CENTER)))
              = UPPER(LTRIM(RTRIM(t.RCNTR)))
            AND h.LEV02 = 'NGRBT'
            AND h.LEV03_DESC =
                'DS Strategic Deterrent Systems'
      )
)

SELECT TOP (60)
    GL AS GL_Account,
    MAX(Description) AS Description,

    COUNT(*) AS Transaction_Rows,
    COUNT(DISTINCT Cost_Center) AS Facilities,

    SUM(Amount) AS Net_Amount,

    SUM(CASE
        WHEN Amount > 0 THEN Amount ELSE 0
    END) AS Positive_Amount,

    SUM(CASE
        WHEN Amount < 0 THEN ABS(Amount) ELSE 0
    END) AS Negative_Absolute

FROM OtherCosts

WHERE GL IS NOT NULL

GROUP BY GL

HAVING SUM(Amount) <> 0

ORDER BY ABS(SUM(Amount)) DESC;
