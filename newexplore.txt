WITH KeyCenters AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS Cost_Center
    FROM ecosystem_source.qmi.costcenterkey
    WHERE NULLIF(LTRIM(RTRIM([Cost Center])), '') IS NOT NULL
),

Original AS (
    SELECT
        COUNT(*) AS Original_Rows,
        SUM(
            TRY_CONVERT(DECIMAL(19,2), [Cost])
        ) AS Original_Total

    FROM ecosystem_source.qmi.controllable_costs

    WHERE TRY_CONVERT(INT, [Year]) = 2026
      AND RIGHT(
          LTRIM(RTRIM(CONVERT(VARCHAR(20), [Quarter]))),
          1
      ) = '1'
),

SAP AS (
    SELECT
        COUNT(*) AS SAP_Rows,

        COUNT(DISTINCT UPPER(LTRIM(RTRIM(t.RCNTR))))
            AS SAP_Cost_Centers,

        SUM(
            CASE
                WHEN TRY_CONVERT(DECIMAL(19,2), t.KSL) > 0
                THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
                ELSE 0
            END
        ) AS Positive_Amount,

        SUM(
            CASE
                WHEN TRY_CONVERT(DECIMAL(19,2), t.KSL) < 0
                THEN ABS(TRY_CONVERT(DECIMAL(19,2), t.KSL))
                ELSE 0
            END
        ) AS Negative_Absolute,

        SUM(
            TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ) AS Net_Amount,

        SUM(
            ABS(TRY_CONVERT(DECIMAL(19,2), t.KSL))
        ) AS Gross_Activity

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND EXISTS (
          SELECT 1
          FROM KeyCenters k
          WHERE UPPER(LTRIM(RTRIM(t.RCNTR)))
                = k.Cost_Center
      )
)

SELECT
    'Q1 2026' AS Period,

    o.Original_Total,
    s.Positive_Amount,
    s.Negative_Absolute,
    s.Net_Amount,
    s.Gross_Activity,

    s.Positive_Amount - o.Original_Total
        AS Positive_Difference,

    s.Gross_Activity - o.Original_Total
        AS Gross_Difference,

    o.Original_Rows,
    s.SAP_Rows,
    s.SAP_Cost_Centers

FROM Original o
CROSS JOIN SAP s;
