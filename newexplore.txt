SET NOCOUNT ON;

/* ============================================================
   CWI Q1 2026 — SYSTEMATIC GL SCOPE DIFFERENCES

   Uses only facilities confidently matched between old and new.

   Purpose:
   Rank GL accounts by their total old-vs-new discrepancy across
   the matched facility population.

   This distinguishes:
     BOTH        — appears in old and new
     OLD_ONLY_GL — CRE report included it; new selection doesn't
     NEW_ONLY_GL — new selection includes it; CRE report didn't

   NO TEMP TABLES.
   ============================================================ */

;WITH FacilityAlias AS (
    SELECT *
    FROM (VALUES
        ('ONE AVIATION WAY',          '25167 ONE AVIATION WAY'),
        ('301 VOYAGER WAY NW',        '301 VOYAGER WAY NORTH WEST'),
        ('4065 COLONEL GLENN',        '4065 COLONEL GLENN HWY'),
        ('STRAWBERRY POINT',          '2828 STRAWBERRY POINT RD'),
        ('FAC WR PARK PLC',           '200 PARK PLACE DR'),
        ('FACILITY - WR IGNICO',      '98 IGNICO DR'),
        ('2300 DISCOVERY DR FL',      '2300 DISCOVERY DR'),
        ('8220 SAN PEDRO DR',         '8220 SAN PEDRO SUITE 300'),
        ('75 WEST ST WALPOLE',        '75 WEST ST')
    ) A (OLD_ADDRESS_KEY, NEW_ADDRESS_KEY)
),

OldRaw AS (
    SELECT
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address])))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS RAW_FACILITY,

        TRY_CONVERT(
            bigint,
            TRY_CONVERT(
                decimal(38,6),
                NULLIF(
                    REPLACE(
                        LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                        ',',
                        ''
                    ),
                    ''
                )
            )
        ) AS GL,

        LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category])))
            AS COST_CATEGORY,

        LTRIM(RTRIM(CONVERT(nvarchar(500), [Cost Element Description])))
            AS GL_DESCRIPTION,

        TRY_CONVERT(decimal(38,2), [Cost])
            AS COST

    FROM ecosystem_source.qmi.controllable_costs

    WHERE TRY_CONVERT(int, [Year]) = 2026
      AND UPPER(
            LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))
          ) = 'Q1'

      AND LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category]))) IN (
          '1 Labor',
          '2 Rent - Land & Building',
          '3 Depreciation & LHI',
          '4 Property Taxes',
          '5 Insurance',
          '6 Maintenance & Repairs',
          '7 Services',
          '8 Utilities'
      )
),

OldRows AS (
    SELECT
        COALESCE(
            A.NEW_ADDRESS_KEY,
            O.RAW_FACILITY
        ) AS FACILITY,

        O.GL,
        O.COST_CATEGORY,
        O.GL_DESCRIPTION,
        O.COST

    FROM OldRaw O

    LEFT JOIN FacilityAlias A
      ON A.OLD_ADDRESS_KEY = O.RAW_FACILITY
),

NewRows AS (
    SELECT
        REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
            UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), facility)))),
            '.', ''),
            ',', ''),
            ' AVENUE', ' AVE'),
            ' ROAD', ' RD'),
            ' DRIVE', ' DR'),
            ' STREET', ' ST'),
            ' HIGHWAY', ' HWY'),
            ' PARKWAY', ' PKWY')
            AS FACILITY,

        TRY_CONVERT(bigint, gl_account_cost_element)
            AS GL,

        LTRIM(RTRIM(cost_category))
            AS COST_CATEGORY,

        LTRIM(RTRIM(cost_element_description))
            AS GL_DESCRIPTION,

        TRY_CONVERT(decimal(38,2), cost)
            AS COST

    FROM ecosystem_source.qmi.controllable_costs_new_extract

    WHERE [year] = 2026
      AND [month] BETWEEN 1 AND 3
      AND division = 'DS C2 & Weapons Integration'
),

OldFacilities AS (
    SELECT DISTINCT FACILITY
    FROM OldRows
),

NewFacilities AS (
    SELECT DISTINCT FACILITY
    FROM NewRows
),

CommonFacilities AS (
    SELECT O.FACILITY
    FROM OldFacilities O
    JOIN NewFacilities N
      ON N.FACILITY = O.FACILITY
),

OldGL AS (
    SELECT
        O.FACILITY,
        O.GL,

        MAX(O.COST_CATEGORY)
            AS OLD_CATEGORY,

        MAX(NULLIF(O.GL_DESCRIPTION, ''))
            AS OLD_DESCRIPTION,

        SUM(O.COST)
            AS OLD_COST

    FROM OldRows O

    JOIN CommonFacilities C
      ON C.FACILITY = O.FACILITY

    WHERE O.GL IS NOT NULL

    GROUP BY
        O.FACILITY,
        O.GL
),

NewGL AS (
    SELECT
        N.FACILITY,
        N.GL,

        MAX(N.COST_CATEGORY)
            AS NEW_CATEGORY,

        MAX(NULLIF(N.GL_DESCRIPTION, ''))
            AS NEW_DESCRIPTION,

        SUM(N.COST)
            AS NEW_COST

    FROM NewRows N

    JOIN CommonFacilities C
      ON C.FACILITY = N.FACILITY

    WHERE N.GL IS NOT NULL

    GROUP BY
        N.FACILITY,
        N.GL
),

FacilityGLComparison AS (
    SELECT
        COALESCE(O.FACILITY, N.FACILITY)
            AS FACILITY,

        COALESCE(O.GL, N.GL)
            AS GL,

        O.OLD_CATEGORY,
        N.NEW_CATEGORY,

        O.OLD_DESCRIPTION,
        N.NEW_DESCRIPTION,

        CASE WHEN O.GL IS NOT NULL THEN 1 ELSE 0 END
            AS EXISTS_OLD,

        CASE WHEN N.GL IS NOT NULL THEN 1 ELSE 0 END
            AS EXISTS_NEW,

        COALESCE(O.OLD_COST, 0)
            AS OLD_COST,

        COALESCE(N.NEW_COST, 0)
            AS NEW_COST,

        COALESCE(N.NEW_COST, 0)
          - COALESCE(O.OLD_COST, 0)
            AS DIFFERENCE

    FROM OldGL O

    FULL OUTER JOIN NewGL N
      ON N.FACILITY = O.FACILITY
     AND N.GL = O.GL
),

GLSummary AS (
    SELECT
        GL,

        MAX(OLD_CATEGORY)
            AS OLD_CATEGORY,

        MAX(NEW_CATEGORY)
            AS NEW_CATEGORY,

        MAX(OLD_DESCRIPTION)
            AS OLD_DESCRIPTION,

        MAX(NEW_DESCRIPTION)
            AS NEW_DESCRIPTION,

        COUNT(*)
            AS FACILITY_COUNT,

        SUM(
            CASE
                WHEN EXISTS_OLD = 1
                 AND EXISTS_NEW = 1
                    THEN 1
                ELSE 0
            END
        ) AS BOTH_FACILITY_COUNT,

        SUM(
            CASE
                WHEN EXISTS_OLD = 1
                 AND EXISTS_NEW = 0
                    THEN 1
                ELSE 0
            END
        ) AS OLD_ONLY_FACILITY_COUNT,

        SUM(
            CASE
                WHEN EXISTS_OLD = 0
                 AND EXISTS_NEW = 1
                    THEN 1
                ELSE 0
            END
        ) AS NEW_ONLY_FACILITY_COUNT,

        SUM(OLD_COST)
            AS OLD_COST,

        SUM(NEW_COST)
            AS NEW_COST,

        SUM(DIFFERENCE)
            AS NET_DIFFERENCE,

        SUM(ABS(DIFFERENCE))
            AS ABS_FACILITY_DIFFERENCE

    FROM FacilityGLComparison

    GROUP BY GL
)

SELECT
    GL,

    CASE
        WHEN OLD_ONLY_FACILITY_COUNT > 0
         AND NEW_ONLY_FACILITY_COUNT = 0
         AND BOTH_FACILITY_COUNT = 0
            THEN 'OLD_ONLY_GL'

        WHEN NEW_ONLY_FACILITY_COUNT > 0
         AND OLD_ONLY_FACILITY_COUNT = 0
         AND BOTH_FACILITY_COUNT = 0
            THEN 'NEW_ONLY_GL'

        WHEN BOTH_FACILITY_COUNT > 0
         AND OLD_ONLY_FACILITY_COUNT = 0
         AND NEW_ONLY_FACILITY_COUNT = 0
            THEN 'BOTH'

        ELSE 'MIXED'
    END AS GL_STATUS,

    OLD_CATEGORY,
    NEW_CATEGORY,

    OLD_DESCRIPTION,
    NEW_DESCRIPTION,

    FACILITY_COUNT,
    BOTH_FACILITY_COUNT,
    OLD_ONLY_FACILITY_COUNT,
    NEW_ONLY_FACILITY_COUNT,

    OLD_COST,
    NEW_COST,

    NET_DIFFERENCE,

    ABS_FACILITY_DIFFERENCE

FROM GLSummary

ORDER BY
    ABS(NET_DIFFERENCE) DESC;
