SET NOCOUNT ON;


/* ============================================================
   RESULT 1 — 3200 Samson Way overall Q1 comparison
   ============================================================ */

;WITH OldData AS (
    SELECT
        TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(
                DECIMAL(38,6),
                NULLIF(
                    REPLACE(
                        LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                        ',',
                        ''
                    ),
                    ''
                )
            )
        ) AS GL,

        LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category])))
            AS COST_CATEGORY,

        TRY_CONVERT(decimal(38,2), [Cost]) AS COST

    FROM ecosystem_source.qmi.controllable_costs

    WHERE TRY_CONVERT(int, [Year]) = 2026
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))) = 'Q1'
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address]))))
            = '3200 SAMSON WAY'
),

NewData AS (
    SELECT
        TRY_CONVERT(BIGINT, gl_account_cost_element) AS GL,
        cost_category AS COST_CATEGORY,
        TRY_CONVERT(decimal(38,2), cost) AS COST

    FROM ecosystem_source.qmi.controllable_costs_new_extract

    WHERE [year] = 2026
      AND [month] BETWEEN 1 AND 3
      AND division = 'DS C2 & Weapons Integration'
      AND UPPER(LTRIM(RTRIM(facility))) = '3200 SAMSON WAY'
)

SELECT
    (SELECT SUM(COST) FROM OldData)
        AS LEGACY_Q1_COST,

    (SELECT SUM(COST) FROM NewData)
        AS NEW_Q1_COST,

    (SELECT SUM(COST) FROM NewData)
      - (SELECT SUM(COST) FROM OldData)
        AS DIFFERENCE,

    CAST(
        100.0 *
        (
            (SELECT SUM(COST) FROM NewData)
          - (SELECT SUM(COST) FROM OldData)
        )
        / NULLIF(
            ABS((SELECT SUM(COST) FROM OldData)),
            0
        )
        AS decimal(12,2)
    ) AS DIFFERENCE_PCT,

    (SELECT SUM(COST)
     FROM OldData
     WHERE GL IS NULL)
        AS LEGACY_COST_WITHOUT_GL,

    (SELECT SUM(COST)
     FROM OldData
     WHERE GL IS NOT NULL)
        AS LEGACY_COST_WITH_GL;


/* ============================================================
   RESULT 2 — GL-by-GL comparison

   BOTH        = same GL appears in old and new
   OLD_ONLY_GL = old has it, new does not
   NEW_ONLY_GL = new has it, old does not

   Legacy rows with no GL are shown separately at the bottom.
   ============================================================ */

;WITH OldRows AS (
    SELECT
        TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(
                DECIMAL(38,6),
                NULLIF(
                    REPLACE(
                        LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                        ',',
                        ''
                    ),
                    ''
                )
            )
        ) AS GL,

        LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category])))
            AS COST_CATEGORY,

        LTRIM(RTRIM(CONVERT(nvarchar(500), [Cost Element Description])))
            AS GL_DESCRIPTION,

        TRY_CONVERT(decimal(38,2), [Cost])
            AS COST

    FROM ecosystem_source.qmi.controllable_costs

    WHERE TRY_CONVERT(int, [Year]) = 2026
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))) = 'Q1'
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address]))))
            = '3200 SAMSON WAY'
),

OldGL AS (
    SELECT
        GL,
        MAX(COST_CATEGORY) AS OLD_CATEGORY,
        MAX(NULLIF(GL_DESCRIPTION, '')) AS OLD_DESCRIPTION,
        COUNT(*) AS OLD_ROW_COUNT,
        SUM(COST) AS OLD_COST
    FROM OldRows
    WHERE GL IS NOT NULL
    GROUP BY GL
),

NewGL AS (
    SELECT
        TRY_CONVERT(BIGINT, gl_account_cost_element) AS GL,
        MAX(cost_category) AS NEW_CATEGORY,
        MAX(NULLIF(cost_element_description, '')) AS NEW_DESCRIPTION,
        COUNT(*) AS NEW_EXTRACT_ROW_COUNT,
        COUNT(DISTINCT cost_center) AS NEW_COST_CENTER_COUNT,
        SUM(TRY_CONVERT(decimal(38,2), cost)) AS NEW_COST

    FROM ecosystem_source.qmi.controllable_costs_new_extract

    WHERE [year] = 2026
      AND [month] BETWEEN 1 AND 3
      AND division = 'DS C2 & Weapons Integration'
      AND UPPER(LTRIM(RTRIM(facility))) = '3200 SAMSON WAY'

    GROUP BY
        TRY_CONVERT(BIGINT, gl_account_cost_element)
),

Comparison AS (
    SELECT
        CASE
            WHEN O.GL IS NOT NULL AND N.GL IS NOT NULL THEN 'BOTH'
            WHEN O.GL IS NOT NULL THEN 'OLD_ONLY_GL'
            ELSE 'NEW_ONLY_GL'
        END AS MATCH_STATUS,

        COALESCE(O.GL, N.GL) AS GL,

        O.OLD_CATEGORY,
        N.NEW_CATEGORY,

        O.OLD_DESCRIPTION,
        N.NEW_DESCRIPTION,

        COALESCE(O.OLD_ROW_COUNT, 0) AS OLD_ROW_COUNT,
        COALESCE(N.NEW_EXTRACT_ROW_COUNT, 0) AS NEW_ROW_COUNT,
        COALESCE(N.NEW_COST_CENTER_COUNT, 0) AS NEW_COST_CENTER_COUNT,

        COALESCE(O.OLD_COST, 0) AS OLD_COST,
        COALESCE(N.NEW_COST, 0) AS NEW_COST,

        COALESCE(N.NEW_COST, 0)
          - COALESCE(O.OLD_COST, 0)
            AS DIFFERENCE

    FROM OldGL O

    FULL OUTER JOIN NewGL N
      ON N.GL = O.GL
)

SELECT
    MATCH_STATUS,
    GL,
    OLD_CATEGORY,
    NEW_CATEGORY,
    OLD_DESCRIPTION,
    NEW_DESCRIPTION,
    OLD_ROW_COUNT,
    NEW_ROW_COUNT,
    NEW_COST_CENTER_COUNT,
    OLD_COST,
    NEW_COST,
    DIFFERENCE,

    CAST(
        CASE
            WHEN OLD_COST = 0 THEN NULL
            ELSE 100.0 * DIFFERENCE / ABS(OLD_COST)
        END
        AS decimal(12,2)
    ) AS DIFFERENCE_PCT

FROM Comparison

ORDER BY
    ABS(DIFFERENCE) DESC;


/* ============================================================
   RESULT 3 — Legacy rows with NO GL

   These cannot be matched directly to SAP RACCT and may explain
   why category totals differ even when the SAP GL selection is right.
   ============================================================ */

;WITH OldRows AS (
    SELECT
        LTRIM(RTRIM(CONVERT(nvarchar(255), [Cost Category])))
            AS COST_CATEGORY,

        LTRIM(RTRIM(CONVERT(nvarchar(500), [Cost Element Description])))
            AS COST_ELEMENT_DESCRIPTION,

        TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(
                DECIMAL(38,6),
                NULLIF(
                    REPLACE(
                        LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                        ',',
                        ''
                    ),
                    ''
                )
            )
        ) AS GL,

        TRY_CONVERT(decimal(38,2), [Cost])
            AS COST

    FROM ecosystem_source.qmi.controllable_costs

    WHERE TRY_CONVERT(int, [Year]) = 2026
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))) = 'Q1'
      AND UPPER(LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address]))))
            = '3200 SAMSON WAY'
)

SELECT
    COST_CATEGORY,
    COUNT(*) AS LEGACY_ROW_COUNT,
    SUM(COST) AS LEGACY_COST_WITHOUT_GL

FROM OldRows

WHERE GL IS NULL

GROUP BY COST_CATEGORY

ORDER BY
    ABS(SUM(COST)) DESC;

/* ============================================================
   RESULT 5 — Raw NEW SAP rows for Samson, Q1

   This bypasses the extract and goes back to
   rb_CVG_Transaction_Details_03.

   Facility is assigned exactly the same way as production:
       RCNTR -> qmi.costcenterkey

   Same selected GL list as production.
   ============================================================ */

;WITH CWIHierarchy AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS COST_CENTER,

        LTRIM(RTRIM(LEV03_DESC))
            AS DIVISION,

        LTRIM(RTRIM(LEV04_DESC))
            AS BUSINESS_UNIT,

        ROW_NUMBER() OVER (
            PARTITION BY UPPER(LTRIM(RTRIM(COST_CENTER)))
            ORDER BY
                last_modified_date DESC,
                created_date DESC,
                id DESC
        ) AS RN

    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy

    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
      AND LTRIM(RTRIM(LEV03_DESC))
            = 'DS C2 & Weapons Integration'
),

FacilityKey AS (
    SELECT
        UPPER(LTRIM(RTRIM([Cost Center])))
            AS COST_CENTER,

        MAX(NULLIF(LTRIM(RTRIM([Address])), ''))
            AS ADDRESS

    FROM ecosystem_source.qmi.costcenterkey

    WHERE NULLIF(LTRIM(RTRIM([Cost Center])), '') IS NOT NULL

    GROUP BY
        UPPER(LTRIM(RTRIM([Cost Center])))
),

SAPRows AS (
    SELECT
        TRY_CONVERT(int, T.GJAHR) AS [YEAR],
        TRY_CONVERT(int, T.POPER) AS [MONTH],

        H.BUSINESS_UNIT,

        UPPER(LTRIM(RTRIM(T.RCNTR)))
            AS COST_CENTER,

        K.ADDRESS AS FACILITY,

        TRY_CONVERT(bigint, T.RACCT)
            AS GL,

        NULLIF(LTRIM(RTRIM(T.GL_TXT20)), '')
            AS GL_DESCRIPTION,

        TRY_CONVERT(decimal(19,2), T.KSL)
            AS COST

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 T

    JOIN CWIHierarchy H
      ON H.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))
     AND H.RN = 1

    JOIN FacilityKey K
      ON K.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))

    WHERE TRY_CONVERT(int, T.GJAHR) = 2026
      AND TRY_CONVERT(int, T.POPER) BETWEEN 1 AND 3

      AND UPPER(LTRIM(RTRIM(K.ADDRESS)))
            = '3200 SAMSON WAY'

      AND (
          TRY_CONVERT(bigint, T.RACCT)
              BETWEEN 4100000 AND 4109999

          OR TRY_CONVERT(bigint, T.RACCT) IN (
              8320037,

              4300591, 4300625, 4300887, 4301077,

              4301048, 4301049, 4301050, 4301231,

              4300626,

              4300852, 4300630, 8320479,

              4300884, 4300699, 4300702, 4300704,
              4300705, 4300701, 4300700, 4300671,
              4300643, 4300305, 4300669, 4300698,
              4300680,

              4300703, 4300805, 4300666, 4300807,

              4301102, 4301013, 4301104, 4300628
          )
      )

      AND TRY_CONVERT(decimal(19,2), T.KSL) IS NOT NULL
)

SELECT
    [MONTH],
    BUSINESS_UNIT,
    COST_CENTER,
    FACILITY,
    GL,
    GL_DESCRIPTION,
    COST

FROM SAPRows

ORDER BY
    GL,
    [MONTH],
    COST_CENTER,
    ABS(COST) DESC;
