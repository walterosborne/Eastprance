-- Top Defense Systems roster sites by absolute hours for October 2026.
-- Period starts with the two-digit month and ends with the four-digit year.
WITH labor_by_person AS (
    SELECT
        UPPER(LTRIM(RTRIM(a.MYID))) AS my_id,
        SUM(ABS(TRY_CONVERT(decimal(18, 2), a.Hours))) AS absolute_hours,
        COUNT_BIG(*) AS labor_rows
    FROM [DTO_Business_Management].[rpt].[rb_Actuals_RM_Load_Table] AS a
    WHERE a.[Period] LIKE N'10%2026'
      AND NULLIF(LTRIM(RTRIM(a.MYID)), '') IS NOT NULL
      AND TRY_CONVERT(decimal(18, 2), a.Hours) IS NOT NULL
      AND EXISTS (
          SELECT 1
          FROM [DTO_Business_Management].[rpt].[rb_load_cost_center_hierarchy] AS h
          WHERE LTRIM(RTRIM(h.LEV02)) = 'NGRBT'
            AND UPPER(LTRIM(RTRIM(h.COST_CENTER))) = UPPER(LTRIM(RTRIM(a.Cost_Center)))
            AND NULLIF(LTRIM(RTRIM(h.COST_CENTER)), '') IS NOT NULL
      )
    GROUP BY UPPER(LTRIM(RTRIM(a.MYID)))
),
matched_people AS (
    SELECT
        a.my_id,
        MIN(site.facility) AS facility,
        MAX(a.absolute_hours) AS absolute_hours,
        MAX(a.labor_rows) AS labor_rows
    FROM [DTO_Business_Management].[dbo].[rb_roster_transform] AS r
    CROSS APPLY (VALUES
        (r.Employee_MyID_NSR),
        (r.Employee_MyID_SR),
        (r.Employee_MyID_DM),
        (r.Employee_MyID_SAP)
    ) AS ids(my_id)
    INNER JOIN labor_by_person AS a ON a.my_id = UPPER(LTRIM(RTRIM(ids.my_id)))
    CROSS APPLY (VALUES (
        CONCAT_WS(', ', LTRIM(RTRIM(r.Work_City_NSR)),
            NULLIF(LTRIM(RTRIM(r.Work_State_NSR)), ''))
    )) AS site(facility)
    WHERE UPPER(LTRIM(RTRIM(r.Sector))) IN ('DS', 'DEFENSE SYSTEMS')
      AND NULLIF(LTRIM(RTRIM(r.Work_City_NSR)), '') IS NOT NULL
      AND UPPER(LTRIM(RTRIM(r.Work_City_NSR))) NOT IN ('UNKNOWN CITY', 'UNKNOWN', 'N/A')
    GROUP BY a.my_id
    HAVING MIN(site.facility) = MAX(site.facility)
)
SELECT TOP (30)
    facility,
    SUM(absolute_hours) AS absolute_hours,
    SUM(labor_rows) AS labor_rows,
    COUNT_BIG(*) AS employees
FROM matched_people
GROUP BY facility
ORDER BY absolute_hours DESC, facility;
