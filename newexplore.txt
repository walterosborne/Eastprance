WITH SDS_Hierarchy AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS CC,
        MAX(CC_DESCRIPTION) AS Cost_Center_Name
    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy
    WHERE LEV02 = 'NGRBT'
      AND LEV03_DESC = 'DS Strategic Deterrent Systems'
    GROUP BY UPPER(LTRIM(RTRIM(COST_CENTER)))
),

Mapped AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS CC
    FROM ecosystem_source.qmi.costcenterkey
),

Allocations AS (
    SELECT
        UPPER(LTRIM(RTRIM(t.RCNTR))) AS CC,
        h.Cost_Center_Name,

        CASE
            WHEN h.CC IN (
                'TDA87FC0',
                'TDA87FO0',
                'TDB84FAC',
                'TDB84FCO',
                'TDA87FL0',
                'TDA10H1C'
            )
                THEN 'Central SDS'

            WHEN k.CC IS NOT NULL
                THEN 'Mapped physical facilities'

            ELSE 'Other SDS centers'
        END AS Location_Bucket,

        TRY_CONVERT(DECIMAL(19,2), t.KSL)
            AS Amount

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    JOIN SDS_Hierarchy h
        ON UPPER(LTRIM(RTRIM(t.RCNTR))) = h.CC

    LEFT JOIN Mapped k
        ON h.CC = k.CC

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND TRY_CONVERT(BIGINT, t.RACCT) = 8320563
)

SELECT
    CASE
        WHEN GROUPING(Location_Bucket) = 1
            THEN 'ALL SDS'

        WHEN GROUPING(CC) = 1
            THEN Location_Bucket + ' TOTAL'

        ELSE Location_Bucket
    END AS Location_Bucket,

    CASE
        WHEN GROUPING(CC) = 1
            THEN NULL
        ELSE CC
    END AS Cost_Center,

    CASE
        WHEN GROUPING(CC) = 1
            THEN NULL
        ELSE Cost_Center_Name
    END AS Cost_Center_Name,

    COUNT(*) AS Transaction_Rows,

    COUNT(DISTINCT CC) AS Cost_Centers,

    SUM(Amount) AS Net_Amount,

    SUM(CASE
        WHEN Amount > 0 THEN Amount
        ELSE 0
    END) AS Positive_Amount,

    SUM(CASE
        WHEN Amount < 0 THEN -Amount
        ELSE 0
    END) AS Negative_Absolute

FROM Allocations

GROUP BY GROUPING SETS (
    (Location_Bucket, CC, Cost_Center_Name),
    (Location_Bucket),
    ()
)

HAVING
       GROUPING(CC) = 1
    OR ABS(SUM(Amount)) >= 25000

ORDER BY
    GROUPING(Location_Bucket) DESC,
    GROUPING(CC) DESC,
    ABS(SUM(Amount)) DESC;
