WITH Priscilla AS (
    SELECT
        TRY_CONVERT(INT, [Year]) AS Yr,
        TRY_CONVERT(INT, REPLACE([Quarter], 'Q', '')) AS Qtr,

        SUM(CASE
            WHEN [Cost Category] = '0 Other'
            THEN TRY_CONVERT(DECIMAL(19,2), [Cost])
            ELSE 0
        END) AS Other_Cost,

        SUM(CASE
            WHEN [Cost Category] = '1 Labor'
            THEN TRY_CONVERT(DECIMAL(19,2), [Cost])
            ELSE 0
        END) AS Labor_Cost

    FROM ecosystem_source.qmi.controllable_costs

    WHERE [Address] =
        'Strategic Deterrent Facility/Operations'

    GROUP BY
        TRY_CONVERT(INT, [Year]),
        TRY_CONVERT(INT, REPLACE([Quarter], 'Q', ''))
),

SAP AS (
    SELECT
        TRY_CONVERT(INT, GJAHR) AS Yr,

        CEILING(
            TRY_CONVERT(DECIMAL(10,2), POPER) / 3.0
        ) AS Qtr,

        SUM(CASE
            WHEN TRY_CONVERT(BIGINT, RACCT) = 4301049
            THEN TRY_CONVERT(DECIMAL(19,2), KSL)
            ELSE 0
        END) AS Machinery_Depreciation,

        SUM(CASE
            WHEN TRY_CONVERT(BIGINT, RACCT) = 4301050
            THEN TRY_CONVERT(DECIMAL(19,2), KSL)
            ELSE 0
        END) AS LHI_Amortization

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03

    WHERE LTRIM(RTRIM(RCNTR)) = 'TDA10H1C'
      AND TRY_CONVERT(INT, POPER) BETWEEN 1 AND 12
      AND TRY_CONVERT(BIGINT, RACCT)
          IN (4301049, 4301050)

    GROUP BY
        TRY_CONVERT(INT, GJAHR),

        CEILING(
            TRY_CONVERT(DECIMAL(10,2), POPER) / 3.0
        )
)

SELECT
    p.Yr AS [Year],
    CONCAT('Q', p.Qtr) AS [Quarter],

    p.Other_Cost AS Priscilla_Other,
    s.Machinery_Depreciation,
    s.LHI_Amortization,

    s.Machinery_Depreciation
        + s.LHI_Amortization
        AS SAP_Depreciation_LHI,

    p.Other_Cost
        - s.Machinery_Depreciation
        - s.LHI_Amortization
        AS Difference,

    p.Labor_Cost AS Priscilla_Central_Labor

FROM Priscilla p

JOIN SAP s
    ON p.Yr = s.Yr
   AND p.Qtr = s.Qtr

ORDER BY p.Yr, p.Qtr;
