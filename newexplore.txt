WITH SourceRows AS (

    -- Priscilla's intermediary ERP report
    SELECT
        'ERP' AS Source,
        UPPER(LTRIM(RTRIM(e.[Cost Center]))) AS Cost_Center,
        TRY_CONVERT(BIGINT, e.[G/L Account]) AS GL,
        TRY_CONVERT(DECIMAL(19,2), e.[Actual Amount])
            AS Amount

    FROM ecosystem_source.dbo.erpdata e

    WHERE RIGHT(
        LTRIM(RTRIM(e.[Fiscal Year Period])), 4
    ) = '2026'

      AND TRY_CONVERT(
          INT,
          LEFT(LTRIM(RTRIM(e.[Fiscal Year Period])), 3)
      ) BETWEEN 1 AND 3

      AND UPPER(LTRIM(RTRIM(e.[Cost Center]))) IN (
          'TDA87FC0',
          'TDA87FO0',
          'TDB84FAC',
          'TDB84FCO',
          'TDA87FL0',
          'TDA10H1C'
      )

    UNION ALL

    -- Raw SAP transactions
    SELECT
        'SAP' AS Source,
        UPPER(LTRIM(RTRIM(t.RCNTR))) AS Cost_Center,
        TRY_CONVERT(BIGINT, t.RACCT) AS GL,
        TRY_CONVERT(DECIMAL(19,2), t.KSL)
            AS Amount

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND UPPER(LTRIM(RTRIM(t.RCNTR))) IN (
          'TDA87FC0',
          'TDA87FO0',
          'TDB84FAC',
          'TDB84FCO',
          'TDA87FL0',
          'TDA10H1C'
      )
)

SELECT
    Source,

    CASE
        WHEN GROUPING(Cost_Center) = 1
        THEN 'TOTAL'
        ELSE Cost_Center
    END AS Cost_Center,

    COUNT(*) AS Row_Count,

    COUNT(DISTINCT GL) AS GL_Accounts,

    SUM(Amount) AS All_Accounts_Net,

    SUM(CASE
        WHEN GL BETWEEN 4100000 AND 4109999
          OR GL = 8320037
        THEN Amount
        ELSE 0
    END) AS Labor_Candidates,

    SUM(CASE
        WHEN GL = 8320563
        THEN Amount
        ELSE 0
    END) AS Facility_Labor_Allocation,

    SUM(CASE
        WHEN GL NOT BETWEEN 4100000 AND 4109999
         AND GL NOT IN (8320037, 8320563)
        THEN Amount
        ELSE 0
    END) AS Other_Accounts

FROM SourceRows

GROUP BY GROUPING SETS (
    (Source, Cost_Center),
    (Source)
)

ORDER BY
    Source,
    GROUPING(Cost_Center),
    Cost_Center;
