/* ============================================================
   Collapse old rows onto their bridged new facility key.
   ============================================================ */

WITH LegacyBridged AS (
    SELECT
        COALESCE(A.NEW_ADDRESS_KEY, L.ADDRESS_KEY) AS ADDRESS_KEY,
        L.COST_CATEGORY,
        SUM(L.LEGACY_COST) AS LEGACY_COST
    FROM #LegacyFacilityCategory L

    LEFT JOIN #FacilityAlias A
      ON A.OLD_ADDRESS_KEY = L.ADDRESS_KEY

    GROUP BY
        COALESCE(A.NEW_ADDRESS_KEY, L.ADDRESS_KEY),
        L.COST_CATEGORY
),

LegacyFacility AS (
    SELECT
        ADDRESS_KEY,
        SUM(LEGACY_COST) AS LEGACY_COST
    FROM LegacyBridged
    GROUP BY ADDRESS_KEY
),

NewFacility AS (
    SELECT
        ADDRESS_KEY,
        SUM(NEW_COST) AS NEW_COST
    FROM #NewFacilityCategory
    GROUP BY ADDRESS_KEY
),

CommonFacility AS (
    SELECT
        L.ADDRESS_KEY,
        L.LEGACY_COST,
        N.NEW_COST
    FROM LegacyFacility L

    JOIN NewFacility N
      ON N.ADDRESS_KEY = L.ADDRESS_KEY
)


/* ============================================================
   RESULT 1 — Updated total after alias bridge.
   ============================================================ */

SELECT
    COUNT(*) AS COMMON_FACILITY_COUNT,

    SUM(LEGACY_COST)
        AS LEGACY_COMMON_Q1_COST,

    SUM(NEW_COST)
        AS NEW_COMMON_Q1_COST,

    SUM(NEW_COST)
      - SUM(LEGACY_COST)
        AS DIFFERENCE,

    CAST(
        100.0 *
        (
            SUM(NEW_COST)
          - SUM(LEGACY_COST)
        )
        /
        NULLIF(
            ABS(SUM(LEGACY_COST)),
            0
        )
        AS decimal(12,2)
    ) AS DIFFERENCE_PCT

FROM CommonFacility;


/* ============================================================
   RESULT 2 — Updated facility-by-facility comparison.
   ============================================================ */

SELECT
    C.ADDRESS_KEY AS FACILITY,
    C.LEGACY_COST,
    C.NEW_COST,
    C.NEW_COST - C.LEGACY_COST AS DIFFERENCE,

    CAST(
        100.0 *
        (
            C.NEW_COST
          - C.LEGACY_COST
        )
        /
        NULLIF(
            ABS(C.LEGACY_COST),
            0
        )
        AS decimal(12,2)
    ) AS DIFFERENCE_PCT

FROM CommonFacility C

ORDER BY
    ABS(C.NEW_COST - C.LEGACY_COST) DESC;


/* ============================================================
   RESULT 3 — What's still unmatched after bridging?
   ============================================================ */

SELECT
    L.ADDRESS_KEY AS OLD_ONLY_FACILITY,
    L.LEGACY_COST

FROM LegacyFacility L

LEFT JOIN NewFacility N
  ON N.ADDRESS_KEY = L.ADDRESS_KEY

WHERE N.ADDRESS_KEY IS NULL

ORDER BY
    L.LEGACY_COST DESC;


SELECT
    N.ADDRESS_KEY AS NEW_ONLY_FACILITY,
    N.NEW_COST

FROM NewFacility N

LEFT JOIN LegacyFacility L
  ON L.ADDRESS_KEY = N.ADDRESS_KEY

WHERE L.ADDRESS_KEY IS NULL

ORDER BY
    N.NEW_COST DESC;
