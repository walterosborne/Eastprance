-- QMI Customer Escapes: raw history keys and current data-variable values
-- Read-only. Run in [Quality Data Mart] and send the three result grids.
-- The current metric version is 041; V01/V02 in the notes are DATA VARIABLES.
-- Candidate counts are level-4 IDs 148 (software), 151 (hardware),
-- and 154 (documentation). IDs 147/150/153 are the significant subsets.
-- This is exploratory; do not sum both a total and its subset.
USE [Quality Data Mart];

-- 1. Columns and types needed to build the January 2025 onward raw-data join.
SELECT
    o.name AS table_name,
    c.column_id,
    c.name AS column_name,
    t.name AS data_type
FROM sys.columns AS c
JOIN sys.objects AS o ON o.object_id = c.object_id
JOIN sys.schemas AS s ON s.schema_id = o.schema_id
JOIN sys.types AS t ON t.user_type_id = c.user_type_id
WHERE s.name = 'QAF_RawData'
  AND o.name IN (
      'PMDH_Program_MonthYear_Data_History',
      'PCMV_Program_Configuration_Metric_Version',
      'PCMY_Program_Configuration_Month_Year',
      'MYDT_Month_Year_Date',
      'DATV_Data_Variable',
      'METC_Metric',
      'MCVR_Metric_Version',
      'PROG_Program'
  )
ORDER BY o.name, c.column_id;

-- 2. Check the recent QAF data inputs by variable and reporting status.
-- Display Data is a presentation field; a numeric sum here is diagnostic only.
SELECT
    d.[METC-DATV_ID] AS data_variable_id,
    d.[DATV V0x] AS variable_code,
    d.[QAF Reporting Status] AS reporting_status,
    COUNT(*) AS row_count,
    COUNT(DISTINCT d.[PROG_ID]) AS program_count,
    MIN(mo.[Data Month]) AS first_month,
    MAX(mo.[Data Month]) AS last_month,
    COUNT(TRY_CONVERT(decimal(18, 2), d.[Display Data])) AS numeric_display_rows,
    SUM(TRY_CONVERT(decimal(18, 2), d.[Display Data])) AS diagnostic_display_sum
FROM [QAF].[QAF Data] AS d
JOIN [QAF].[QAF Months] AS mo ON mo.[MYDT_ID] = d.[MYDT_ID]
WHERE d.[QAF Metric Level] = 4
  AND d.[METC-DATV_ID] IN (147, 148, 150, 151, 153, 154)
  AND mo.[Data Month] >= '2025-01-01'
GROUP BY d.[METC-DATV_ID], d.[DATV V0x], d.[QAF Reporting Status]
ORDER BY d.[METC-DATV_ID], d.[DATV V0x], d.[QAF Reporting Status];

-- 3. Five recent FINAL examples for each candidate variable.
WITH examples AS (
    SELECT
        mo.[Data Month] AS data_month,
        d.[PROG_ID] AS program_id,
        d.[METC-DATV_ID] AS data_variable_id,
        d.[DATV V0x] AS variable_code,
        d.[Display Data] AS display_data,
        d.[Score Final] AS final_score,
        d.[QAF Reporting Status] AS reporting_status,
        ROW_NUMBER() OVER (
            PARTITION BY d.[METC-DATV_ID]
            ORDER BY mo.[Data Month] DESC, d.[PROG_ID]
        ) AS rn
    FROM [QAF].[QAF Data] AS d
    JOIN [QAF].[QAF Months] AS mo ON mo.[MYDT_ID] = d.[MYDT_ID]
    WHERE d.[QAF Metric Level] = 4
      AND d.[METC-DATV_ID] IN (147, 148, 150, 151, 153, 154)
      AND d.[QAF Reporting Status] = 'FINAL'
)
SELECT
    data_month, program_id, data_variable_id, variable_code,
    display_data, final_score, reporting_status
FROM examples
WHERE rn <= 5
ORDER BY data_variable_id, data_month DESC, program_id;
