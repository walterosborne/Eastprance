-- Inspect the available columns before choosing the labor-to-roster join key.
SELECT TABLE_SCHEMA, TABLE_NAME, COLUMN_NAME, DATA_TYPE
FROM [DTO_Business_Management].INFORMATION_SCHEMA.COLUMNS
WHERE (TABLE_SCHEMA = 'rpt' AND TABLE_NAME IN ('rb_Actuals_RM_Load_Table', 'rb_load_roster'))
   OR (TABLE_SCHEMA = 'dbo' AND TABLE_NAME = 'rb_roster_transform')
ORDER BY
    CASE
        WHEN TABLE_NAME = 'rb_Actuals_RM_Load_Table' THEN 1
        WHEN TABLE_NAME = 'rb_load_roster' THEN 2
        ELSE 3
    END,
    ORDINAL_POSITION;
