WITH SDS AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS Cost_Center,
        MAX(CC_DESCRIPTION) AS Cost_Center_Name
    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy
    WHERE LEV02 = 'NGRBT'
      AND LEV03_DESC = 'DS Strategic Deterrent Systems'
    GROUP BY
        UPPER(LTRIM(RTRIM(COST_CENTER)))
)

SELECT
    Cost_Center,
    Cost_Center_Name
FROM SDS
WHERE EXISTS (
    SELECT 1
    FROM (VALUES
        ('OGDEN'),
        ('KEMP'),
        ('WAREHOUSE'),
        ('SURGE'),
        ('HUNTSVILLE'),
        ('MADISON'),
        ('WYNN'),
        ('CHEYENNE'),
        ('CLEARFIELD'),
        ('UNIVERSITY'),
        ('SANTA'),
        ('MARIA'),
        ('VANDENBERG'),
        ('FAIRWAY'),
        ('FACILITY')
    ) AS SearchTerms(Term)
    WHERE UPPER(Cost_Center_Name)
        LIKE '%' + SearchTerms.Term + '%'
)
ORDER BY Cost_Center_Name, Cost_Center;
