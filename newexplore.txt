WITH ERP AS (
    SELECT
        [Fiscal Year Period] AS Period,
        TRY_CONVERT(BIGINT, [G/L Account]) AS GL,
        COUNT(*) AS ERP_Rows,
        SUM(TRY_CONVERT(DECIMAL(19,2), [Actual Amount])) AS ERP_Net,
        SUM(ABS(TRY_CONVERT(DECIMAL(19,2), [Actual Amount]))) AS ERP_Gross
    FROM ecosystem_source.dbo.erpdata
    WHERE LTRIM(RTRIM([Cost Center])) = 'TDA84LF0'
      AND [Fiscal Year Period] IN (
          '006/2026', '007/2026', '008/2026'
      )
    GROUP BY
        [Fiscal Year Period],
        TRY_CONVERT(BIGINT, [G/L Account])
),
SAP AS (
    SELECT
        RIGHT('000' + LTRIM(RTRIM(POPER)), 3)
            + '/' + LTRIM(RTRIM(GJAHR)) AS Period,
        TRY_CONVERT(BIGINT, RACCT) AS GL,
        COUNT(*) AS SAP_Rows,
        SUM(TRY_CONVERT(DECIMAL(19,2), KSL)) AS SAP_Net,
        SUM(ABS(TRY_CONVERT(DECIMAL(19,2), KSL))) AS SAP_Gross
    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03
    WHERE LTRIM(RTRIM(RCNTR)) = 'TDA84LF0'
      AND TRY_CONVERT(INT, GJAHR) = 2026
      AND TRY_CONVERT(INT, POPER) BETWEEN 6 AND 8
    GROUP BY
        GJAHR,
        POPER,
        TRY_CONVERT(BIGINT, RACCT)
)
SELECT TOP (25)
    COALESCE(E.Period, S.Period) AS Period,
    COALESCE(E.GL, S.GL) AS GL,

    E.ERP_Rows,
    S.SAP_Rows,

    E.ERP_Net,
    S.SAP_Net,

    E.ERP_Gross,
    S.SAP_Gross,

    COALESCE(E.ERP_Gross, 0)
      - COALESCE(S.SAP_Gross, 0) AS Gross_Difference

FROM ERP E
FULL OUTER JOIN SAP S
    ON E.Period = S.Period
   AND E.GL = S.GL

ORDER BY ABS(
    COALESCE(E.ERP_Gross, 0)
      - COALESCE(S.SAP_Gross, 0)
) DESC;
