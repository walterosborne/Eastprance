-- Find fields that could link labor postings to a charge object or labor category.
DESCRIBE TABLE source_systems_dev.sql_central_integration.lfc_transaction_details_s3;

-- Check all accounts with nonzero hour quantities in fiscal 2026.
WITH hour_postings AS (
  SELECT
    NULLIF(TRIM(RACCT), '') AS account,
    TRIM(GL_TXT50) AS description,
    TRY_CAST(NET_MSL AS DECIMAL(24, 4)) AS hours_quantity
  FROM source_systems_dev.sql_central_integration.lfc_transaction_details_s3
  WHERE TRY_CAST(GJAHR AS INT) = 2026
    AND UPPER(TRIM(RUNIT)) = 'H'
)
SELECT
  account,
  description,
  COUNT(*) AS nonzero_quantity_postings,
  SUM(hours_quantity) AS signed_hours,
  SUM(ABS(hours_quantity)) AS absolute_hours
FROM hour_postings
WHERE hours_quantity <> 0
GROUP BY account, description
ORDER BY absolute_hours DESC, account, description;
