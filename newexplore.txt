WITH SDS AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS CC,
        MAX(CC_DESCRIPTION) AS Cost_Center_Name
    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy
    WHERE LEV02 = 'NGRBT'
      AND LEV03_DESC = 'DS Strategic Deterrent Systems'
    GROUP BY UPPER(LTRIM(RTRIM(COST_CENTER)))
),

Unmapped AS (
    SELECT
        h.CC,
        h.Cost_Center_Name
    FROM SDS h
    WHERE NOT EXISTS (
        SELECT 1
        FROM ecosystem_source.qmi.costcenterkey k
        WHERE UPPER(LTRIM(RTRIM(k.[Cost Center]))) = h.CC
    )
)

SELECT TOP (30)
    u.CC AS Cost_Center,
    u.Cost_Center_Name,

    COUNT(*) AS Transaction_Rows,

    SUM(
        TRY_CONVERT(DECIMAL(19,2), t.KSL)
    ) AS Q1_Selected_Cost

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

JOIN Unmapped u
    ON UPPER(LTRIM(RTRIM(t.RCNTR))) = u.CC

WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
  AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

  AND TRY_CONVERT(BIGINT, t.RACCT) IN (
      4300591,  -- Real estate lease
      4300625,  -- Variable rent / CAM
      4301050,  -- LHI amortization
      4301049,  -- Machinery depreciation
      4301102,  -- Electricity
      4300703   -- Janitorial
  )

GROUP BY
    u.CC,
    u.Cost_Center_Name

HAVING SUM(
    TRY_CONVERT(DECIMAL(19,2), t.KSL)
) <> 0

ORDER BY ABS(
    SUM(TRY_CONVERT(DECIMAL(19,2), t.KSL))
) DESC;
