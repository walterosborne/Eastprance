WITH Central AS (
    SELECT
        LTRIM(RTRIM(t.RCNTR)) AS Cost_Center,
        TRY_CONVERT(BIGINT, t.RACCT) AS GL,
        t.GL_TXT20 AS Description,
        TRY_CONVERT(DECIMAL(19,2), t.KSL) AS Amount
    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t
    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3
      AND LTRIM(RTRIM(t.RCNTR)) IN (
          'TDA87FC0',
          'TDA87FO0',
          'TDB84FAC',
          'TDB84FCO',
          'TDA87FL0',
          'TDA10H1C'
      )
),

Classified AS (
    SELECT
        *,
        CASE
            WHEN GL BETWEEN 4100000 AND 4109999
              OR GL = 8320037
                THEN 'LABOR CANDIDATE'

            WHEN GL IN (
                8320563,
                8320578,
                8320584,
                8445003
            )
                THEN 'FACILITY ALLOCATION'
        END AS Account_Group
    FROM Central
)

SELECT
    Account_Group,

    CASE
        WHEN GROUPING(Cost_Center) = 1
            THEN 'TOTAL'
        ELSE Cost_Center
    END AS Cost_Center,

    CASE
        WHEN GROUPING(GL) = 1
            THEN NULL
        ELSE GL
    END AS GL_Account,

    MAX(Description) AS Description,

    COUNT(*) AS Transaction_Rows,

    SUM(Amount) AS Q1_Net,

    SUM(CASE
        WHEN Amount > 0 THEN Amount ELSE 0
    END) AS Positive_Amount,

    SUM(CASE
        WHEN Amount < 0 THEN -Amount ELSE 0
    END) AS Negative_Absolute,

    CASE
        WHEN GROUPING(Cost_Center) = 1
         AND Account_Group = 'LABOR CANDIDATE'
        THEN 1949847.71 - SUM(Amount)
    END AS Difference_From_Priscilla

FROM Classified

WHERE Account_Group IS NOT NULL

GROUP BY GROUPING SETS (
    (Account_Group, Cost_Center, GL),
    (Account_Group)
)

ORDER BY
    Account_Group,
    GROUPING(Cost_Center),
    Cost_Center,
    GL;
