SET NOCOUNT ON;


/* ============================================================
   RESULT 1 — Raw Q1 SAP activity for GL 4300625 at Samson Way
   ============================================================ */

;WITH CWIHierarchy AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS COST_CENTER,
        LTRIM(RTRIM(LEV03_DESC)) AS DIVISION,
        LTRIM(RTRIM(LEV04_DESC)) AS BUSINESS_UNIT,

        ROW_NUMBER() OVER (
            PARTITION BY UPPER(LTRIM(RTRIM(COST_CENTER)))
            ORDER BY
                last_modified_date DESC,
                created_date DESC,
                id DESC
        ) AS RN

    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy

    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
      AND LTRIM(RTRIM(LEV03_DESC))
            = 'DS C2 & Weapons Integration'
),

FacilityKey AS (
    SELECT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS COST_CENTER,
        MAX(NULLIF(LTRIM(RTRIM([Address])), '')) AS ADDRESS

    FROM ecosystem_source.qmi.costcenterkey

    WHERE NULLIF(LTRIM(RTRIM([Cost Center])), '') IS NOT NULL

    GROUP BY
        UPPER(LTRIM(RTRIM([Cost Center])))
)

SELECT
    TRY_CONVERT(int, T.GJAHR) AS [YEAR],
    TRY_CONVERT(int, T.POPER) AS [MONTH],

    H.BUSINESS_UNIT,

    UPPER(LTRIM(RTRIM(T.RCNTR)))
        AS COST_CENTER,

    K.ADDRESS AS FACILITY,

    TRY_CONVERT(bigint, T.RACCT)
        AS GL,

    T.GL_TXT20 AS GL_DESCRIPTION,

    T.ACCT_LEVEL02_TEXT,
    T.ACCT_LEVEL03_TEXT,
    T.ACCT_LEVEL04_TEXT,

    TRY_CONVERT(decimal(19,2), T.KSL)
        AS COST

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 T

JOIN CWIHierarchy H
  ON H.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))
 AND H.RN = 1

JOIN FacilityKey K
  ON K.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))

WHERE TRY_CONVERT(int, T.GJAHR) = 2026
  AND TRY_CONVERT(int, T.POPER) BETWEEN 1 AND 3

  AND UPPER(LTRIM(RTRIM(K.ADDRESS)))
        = '3200 SAMSON WAY'

  AND TRY_CONVERT(bigint, T.RACCT)
        = 4300625

ORDER BY
    TRY_CONVERT(int, T.POPER),
    ABS(TRY_CONVERT(decimal(19,2), T.KSL)) DESC;


/* ============================================================
   RESULT 2 — Monthly totals for 4300625 at Samson
   ============================================================ */

;WITH CWIHierarchy AS (
    SELECT
        UPPER(LTRIM(RTRIM(COST_CENTER))) AS COST_CENTER,

        ROW_NUMBER() OVER (
            PARTITION BY UPPER(LTRIM(RTRIM(COST_CENTER)))
            ORDER BY
                last_modified_date DESC,
                created_date DESC,
                id DESC
        ) AS RN

    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy

    WHERE LTRIM(RTRIM(LEV02)) = 'NGRBT'
      AND LTRIM(RTRIM(LEV03_DESC))
            = 'DS C2 & Weapons Integration'
),

FacilityKey AS (
    SELECT
        UPPER(LTRIM(RTRIM([Cost Center]))) AS COST_CENTER,
        MAX(NULLIF(LTRIM(RTRIM([Address])), '')) AS ADDRESS

    FROM ecosystem_source.qmi.costcenterkey

    GROUP BY
        UPPER(LTRIM(RTRIM([Cost Center])))
)

SELECT
    TRY_CONVERT(int, T.POPER) AS [MONTH],

    COUNT(*) AS SAP_ROW_COUNT,

    SUM(
        TRY_CONVERT(decimal(19,2), T.KSL)
    ) AS GL_4300625_COST

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 T

JOIN CWIHierarchy H
  ON H.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))
 AND H.RN = 1

JOIN FacilityKey K
  ON K.COST_CENTER = UPPER(LTRIM(RTRIM(T.RCNTR)))

WHERE TRY_CONVERT(int, T.GJAHR) = 2026
  AND TRY_CONVERT(int, T.POPER) BETWEEN 1 AND 3

  AND UPPER(LTRIM(RTRIM(K.ADDRESS)))
        = '3200 SAMSON WAY'

  AND TRY_CONVERT(bigint, T.RACCT)
        = 4300625

GROUP BY
    TRY_CONVERT(int, T.POPER)

ORDER BY [MONTH];


/* ============================================================
   RESULT 3 — Does legacy Q1 contain GL 4300625 ANYWHERE?

   If this returns zero rows, the old report simply did not carry
   this GL at all.
   ============================================================ */

SELECT
    [Address],

    [Cost Category],

    [Cost Element],

    [Cost Element Description],

    SUM(
        TRY_CONVERT(decimal(38,2), [Cost])
    ) AS LEGACY_Q1_COST,

    COUNT(*) AS LEGACY_ROW_COUNT

FROM ecosystem_source.qmi.controllable_costs

WHERE TRY_CONVERT(int, [Year]) = 2026

  AND UPPER(
        LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))
      ) = 'Q1'

  AND TRY_CONVERT(
        bigint,
        TRY_CONVERT(
            decimal(38,6),
            NULLIF(
                REPLACE(
                    LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                    ',',
                    ''
                ),
                ''
            )
        )
      ) = 4300625

GROUP BY
    [Address],
    [Cost Category],
    [Cost Element],
    [Cost Element Description]

ORDER BY
    ABS(
        SUM(
            TRY_CONVERT(decimal(38,2), [Cost])
        )
    ) DESC;


/* ============================================================
   RESULT 4 — New CWI Q1 GL 4300625 across ALL facilities

   This tells us whether VarRent CAM is common across the new
   source or just a Samson anomaly.
   ============================================================ */

SELECT
    facility,

    cost_center,

    cost_category,

    gl_account_cost_element,

    cost_element_description,

    SUM(
        TRY_CONVERT(decimal(38,2), cost)
    ) AS NEW_Q1_COST

FROM ecosystem_source.qmi.controllable_costs_new_extract

WHERE [year] = 2026
  AND [month] BETWEEN 1 AND 3

  AND division = 'DS C2 & Weapons Integration'

  AND TRY_CONVERT(bigint, gl_account_cost_element)
        = 4300625

GROUP BY
    facility,
    cost_center,
    cost_category,
    gl_account_cost_element,
    cost_element_description

ORDER BY
    ABS(
        SUM(
            TRY_CONVERT(decimal(38,2), cost)
        )
    ) DESC;


/* ============================================================
   RESULT 5 — Legacy Samson Rent rows

   Confirms exactly what the old CRE report counted as Rent for
   this facility.
   ============================================================ */

SELECT
    [Cost Category],

    [Cost Element],

    [Cost Element Description],

    TRY_CONVERT(decimal(38,2), [Cost])
        AS COST

FROM ecosystem_source.qmi.controllable_costs

WHERE TRY_CONVERT(int, [Year]) = 2026

  AND UPPER(
        LTRIM(RTRIM(CONVERT(nvarchar(100), [Quarter])))
      ) = 'Q1'

  AND UPPER(
        LTRIM(RTRIM(CONVERT(nvarchar(1000), [Address])))
      ) = '3200 SAMSON WAY'

  AND LTRIM(
        RTRIM(CONVERT(nvarchar(255), [Cost Category]))
      ) = '2 Rent - Land & Building'

ORDER BY
    TRY_CONVERT(
        bigint,
        TRY_CONVERT(
            decimal(38,6),
            NULLIF(
                REPLACE(
                    LTRIM(RTRIM(CONVERT(varchar(100), [Cost Element]))),
                    ',',
                    ''
                ),
                ''
            )
        )
    );
