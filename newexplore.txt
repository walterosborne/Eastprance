WITH Original AS (
    SELECT
        TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(DECIMAL(19,2), [Cost Element])
        ) AS GL,
        [Cost Category] AS Category,
        SUM(TRY_CONVERT(DECIMAL(19,2), [Cost]))
            AS Amount
    FROM ecosystem_source.qmi.controllable_costs
    WHERE TRY_CONVERT(INT, [Year]) = 2026
      AND LTRIM(RTRIM([Quarter])) = 'Q1'
      AND TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(DECIMAL(19,2), [Cost Element])
          ) IN (4300852, 4300630, 8320479)
    GROUP BY
        TRY_CONVERT(
            BIGINT,
            TRY_CONVERT(DECIMAL(19,2), [Cost Element])
        ),
        [Cost Category]
)

SELECT
    GL AS GL_Account,
    Category AS Priscilla_Category,
    Amount AS Priscilla_Q1
FROM Original
ORDER BY GL, Category;
