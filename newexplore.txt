SELECT
    CASE
        WHEN TRY_CONVERT(
            DECIMAL(19,2), [Cost Element]
        ) IS NULL
            THEN 'Missing Cost Element'
        ELSE 'Populated Cost Element'
    END AS Cost_Element_Status,

    COUNT(*) AS Report_Rows,

    SUM(
        TRY_CONVERT(DECIMAL(19,2), [Cost])
    ) AS Q1_Insurance_Cost

FROM ecosystem_source.qmi.controllable_costs

WHERE TRY_CONVERT(INT, [Year]) = 2026
  AND LTRIM(RTRIM([Quarter])) = 'Q1'
  AND LTRIM(RTRIM([Cost Category])) = '5 Insurance'

GROUP BY
    CASE
        WHEN TRY_CONVERT(
            DECIMAL(19,2), [Cost Element]
        ) IS NULL
            THEN 'Missing Cost Element'
        ELSE 'Populated Cost Element'
    END;
