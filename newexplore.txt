WITH defense_centers AS (
  SELECT DISTINCT UPPER(TRIM(CAST(new_Cost_Center AS STRING))) AS center
  FROM `finance_dev`.`analytics`.`cost_vis_full`
  WHERE CC_Bus_Unit_Lvl_2 LIKE '%Defense Systems%'
),
compact_centers AS (
  SELECT DISTINCT REGEXP_REPLACE(center, '[^A-Z0-9]', '') AS center
  FROM defense_centers
),
ledger AS (
  SELECT 'S3' AS source, GJAHR, POPER, SEGMENT, RCNTR, KSL
  FROM `raw_landing_dev`.`sql_central_integration`.`src_lfc_transaction_details_s3`
  UNION ALL
  SELECT 'SX', GJAHR, POPER, SEGMENT, RCNTR, KSL
  FROM `raw_landing_dev`.`sql_central_integration`.`src_lfc_transaction_details_sx`
),
monthly AS (
  SELECT source,
    TRY_CAST(GJAHR AS INT) AS year,
    TRY_CAST(POPER AS INT) AS month,
    TRIM(CAST(SEGMENT AS STRING)) AS segment,
    UPPER(TRIM(CAST(RCNTR AS STRING))) AS center,
    TRY_CAST(KSL AS DOUBLE) AS cost
  FROM ledger
)
SELECT m.source, m.year, m.month, m.segment,
  COUNT(*) AS rows,
  COUNT_IF(NULLIF(m.center, '') IS NOT NULL) AS rows_with_rcntr,
  COUNT_IF(d.center IS NOT NULL) AS ds_center_rows,
  COUNT_IF(d.center IS NULL AND c.center IS NOT NULL) AS formatting_only_rows,
  ROUND(SUM(m.cost), 2) AS all_ksl,
  ROUND(SUM(CASE WHEN d.center IS NOT NULL THEN m.cost ELSE 0 END), 2) AS ds_center_ksl
FROM monthly m
LEFT JOIN defense_centers d ON d.center = m.center
LEFT JOIN compact_centers c
  ON c.center = REGEXP_REPLACE(m.center, '[^A-Z0-9]', '')
WHERE m.year >= 2025 AND m.month BETWEEN 1 AND 12
GROUP BY m.source, m.year, m.month, m.segment
ORDER BY m.source, m.year DESC, m.month DESC, m.segment;
