/* ================================================
   RESULT 1: THE THREE ERP LABOR RECORDS
   ================================================ */

SELECT
    *
FROM ecosystem_source.dbo.erpdata e
WHERE UPPER(LTRIM(RTRIM(e.[Cost Center])))
      = 'TDA87FC0'

  AND TRY_CONVERT(BIGINT, e.[G/L Account])
      = 4100000

  AND RIGHT(
      LTRIM(RTRIM(e.[Fiscal Year Period])), 4
  ) = '2026'

  AND TRY_CONVERT(
      INT,
      LEFT(LTRIM(RTRIM(e.[Fiscal Year Period])), 3)
  ) BETWEEN 1 AND 3

ORDER BY e.[Fiscal Year Period];


/* ================================================
   RESULT 2: DOES 4100000 EXIST ANYWHERE IN SAP?
   ================================================ */

SELECT
    COUNT(*) AS SAP_4100000_Rows,

    COUNT(DISTINCT
        UPPER(LTRIM(RTRIM(t.RCNTR)))
    ) AS SAP_Cost_Centers,

    SUM(
        TRY_CONVERT(DECIMAL(19,2), t.KSL)
    ) AS SAP_4100000_Total,

    SUM(CASE
        WHEN UPPER(LTRIM(RTRIM(t.RCNTR)))
             = 'TDA87FC0'
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS SAP_Facility_Ops_4100000

FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
  AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

  AND TRY_CONVERT(BIGINT, t.RACCT) = 4100000;
