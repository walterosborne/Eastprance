SELECT
    [Year],
    [Quarter],
    COUNT(*) AS Row_Count,
    SUM(TRY_CONVERT(DECIMAL(19,2), [Cost])) AS Total_Cost
FROM ecosystem_source.qmi.controllable_costs
WHERE TRY_CONVERT(INT, [Year]) = 2025
   OR (
       TRY_CONVERT(INT, [Year]) = 2026
       AND [Quarter] IN ('Q1', 'Q2')
   )
GROUP BY [Year], [Quarter]
ORDER BY [Year], [Quarter];
