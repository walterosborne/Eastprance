WITH Central AS (
    SELECT
        UPPER(LTRIM(RTRIM(t.RCNTR))) AS Cost_Center,

        COALESCE(
            NULLIF(LTRIM(RTRIM(t.ACCT_LEVEL02_TEXT)), ''),
            '(blank)'
        ) AS Account_Level_2,

        COALESCE(
            NULLIF(LTRIM(RTRIM(t.ACCT_LEVEL03_TEXT)), ''),
            '(blank)'
        ) AS Account_Level_3,

        TRY_CONVERT(BIGINT, t.RACCT) AS GL,

        TRY_CONVERT(DECIMAL(19,2), t.KSL)
            AS Amount

    FROM DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t

    WHERE TRY_CONVERT(INT, t.GJAHR) = 2026
      AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

      AND UPPER(LTRIM(RTRIM(t.RCNTR))) IN (
          'TDA87FC0', -- Facility/Ops
          'TDA87FO0', -- Facility Overhead
          'TDB84FAC',
          'TDB84FCO',
          'TDA87FL0', -- Fleet Management
          'TDA10H1C'  -- Strategic Deterrent Overhead
      )
)

SELECT
    Cost_Center,
    Account_Level_2,
    Account_Level_3,

    COUNT(*) AS Transaction_Rows,
    COUNT(DISTINCT GL) AS GL_Accounts,

    SUM(Amount) AS Q1_Net,

    SUM(
        CASE
            WHEN Amount > 0 THEN Amount
            ELSE 0
        END
    ) AS Positive_Amount,

    SUM(
        CASE
            WHEN Amount < 0 THEN -Amount
            ELSE 0
        END
    ) AS Negative_Absolute

FROM Central

GROUP BY
    Cost_Center,
    Account_Level_2,
    Account_Level_3

ORDER BY
    Cost_Center,
    ABS(SUM(Amount)) DESC;
