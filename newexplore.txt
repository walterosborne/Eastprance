WITH SDS_Addresses AS (
    SELECT DISTINCT
        UPPER(LTRIM(RTRIM([Address]))) AS Address_Key
    FROM ecosystem_source.qmi.costcenterkey
    WHERE UPPER(LTRIM(RTRIM([Cost Center]))) IN (
        'TDA87R10', 'TDA87R20',
        'TDA87R30', 'TDA87R40',
        'TDA87OM0', 'TDA87HN0', 'TDA87H20',
        'TDA87KE0', 'TDA87UK0', 'TDA87SR0',
        'TDA87UV0', 'TDA87WY0', 'TDA87VF0'
    )
)

SELECT
    c.[Address],
    COUNT(*) AS Report_Rows,
    SUM(TRY_CONVERT(DECIMAL(19,2), c.[Cost]))
        AS Q1_Report_Cost

FROM ecosystem_source.qmi.controllable_costs c

WHERE TRY_CONVERT(INT, c.[Year]) = 2026
  AND LTRIM(RTRIM(c.[Quarter])) = 'Q1'

  AND EXISTS (
      SELECT 1
      FROM SDS_Addresses s
      WHERE s.Address_Key =
          UPPER(LTRIM(RTRIM(c.[Address])))
  )

GROUP BY c.[Address]

ORDER BY Q1_Report_Cost DESC;
