WITH Facilities AS (
    SELECT DISTINCT
        LTRIM(RTRIM(COST_CENTER)) AS Cost_Center,
        CC_DESCRIPTION AS Cost_Center_Name
    FROM DTO_Business_Management.rpt.rb_load_cost_center_hierarchy
    WHERE LEV02 = 'NGRBT'
      AND LEV03_DESC = 'DS Strategic Deterrent Systems'
      AND (
          UPPER(CC_DESCRIPTION) LIKE '%CHEYENNE%'
          OR UPPER(CC_DESCRIPTION) LIKE '%888%UNIVERSITY%'
      )
)

SELECT
    f.Cost_Center,
    f.Cost_Center_Name,

    COUNT(t.RCNTR) AS Transaction_Rows,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT)
             IN (4300591, 4300625)
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Rent,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT)
             IN (4301050, 4301049, 4301048)
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Depreciation_LHI,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT) = 4301102
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Electricity,

    SUM(CASE
        WHEN TRY_CONVERT(BIGINT, t.RACCT) = 4300703
        THEN TRY_CONVERT(DECIMAL(19,2), t.KSL)
        ELSE 0
    END) AS Janitorial

FROM Facilities f

LEFT JOIN DTO_Business_Management.src.rb_CVG_Transaction_Details_03 t
    ON LTRIM(RTRIM(t.RCNTR)) = f.Cost_Center
   AND TRY_CONVERT(INT, t.GJAHR) = 2026
   AND TRY_CONVERT(INT, t.POPER) BETWEEN 1 AND 3

GROUP BY
    f.Cost_Center,
    f.Cost_Center_Name

ORDER BY f.Cost_Center_Name, f.Cost_Center;
