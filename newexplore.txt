        LTRIM(RTRIM([Cost Center])),
        ''

        UPPER(LTRIM(RTRIM([Cost Center])))

/* ============================================================
   ONLY COMPARE FACILITIES THAT EXIST IN BOTH DATASETS
   ============================================================ */

LegacyFacilities AS (
    SELECT DISTINCT FACILITY
    FROM Legacy
),

NewFacilities AS (
    SELECT DISTINCT FACILITY
    FROM NewRows
),

CommonFacilities AS (
    SELECT L.FACILITY

    FROM LegacyFacilities L

    JOIN NewFacilities N
      ON N.FACILITY = L.FACILITY
),


/* ============================================================
   LEGACY GL TOTALS
   ============================================================ */

LegacyGL AS (
    SELECT
        L.FACILITY,
        L.GL,

        MAX(L.COST_CATEGORY)
            AS LEGACY_CATEGORY,

        MAX(L.GL_DESCRIPTION)
            AS LEGACY_DESCRIPTION,

        SUM(L.COST)
            AS LEGACY_COST

    FROM Legacy L

    JOIN CommonFacilities C
      ON C.FACILITY = L.FACILITY

    WHERE L.GL IS NOT NULL

    GROUP BY
        L.FACILITY,
        L.GL
),


/* Legacy dollars with no Cost Element */
LegacyNoGL AS (
    SELECT
        L.FACILITY,

        L.COST_CATEGORY
            AS LEGACY_CATEGORY,

        SUM(L.COST)
            AS LEGACY_COST

    FROM Legacy L

    JOIN CommonFacilities C
      ON C.FACILITY = L.FACILITY

    WHERE L.GL IS NULL

    GROUP BY
        L.FACILITY,
        L.COST_CATEGORY
),


/* ============================================================
   NEW GL TOTALS
   ============================================================ */

NewGL AS (
    SELECT
        N.FACILITY,
        N.GL,

        MAX(N.COST_CATEGORY)
            AS NEW_CATEGORY,

        MAX(N.GL_DESCRIPTION)
            AS NEW_DESCRIPTION,

        SUM(N.COST)
            AS NEW_COST

    FROM NewRows N

    JOIN CommonFacilities C
      ON C.FACILITY = N.FACILITY

    GROUP BY
        N.FACILITY,
        N.GL
),


/* ============================================================
   GL COMPARISON
   ============================================================ */

GLComparison AS (
    SELECT
        COALESCE(L.FACILITY, N.FACILITY)
            AS FACILITY,

        COALESCE(L.GL, N.GL)
            AS GL,

        CASE
            WHEN L.GL IS NOT NULL
             AND N.GL IS NOT NULL
                THEN 'BOTH'

            WHEN L.GL IS NOT NULL
                THEN 'LEGACY_ONLY_GL'

            ELSE 'NEW_ONLY_GL'
        END AS MATCH_STATUS,

        L.LEGACY_CATEGORY,
        N.NEW_CATEGORY,

        L.LEGACY_DESCRIPTION,
        N.NEW_DESCRIPTION,

        COALESCE(L.LEGACY_COST, 0)
            AS LEGACY_COST,

        COALESCE(N.NEW_COST, 0)
            AS NEW_COST,

        COALESCE(N.NEW_COST, 0)
          - COALESCE(L.LEGACY_COST, 0)
            AS DIFFERENCE

    FROM LegacyGL L

    FULL OUTER JOIN NewGL N
      ON N.FACILITY = L.FACILITY
     AND N.GL = L.GL
),


/* ============================================================
   INCLUDE LEGACY ROWS THAT HAVE NO GL
   ============================================================ */

ReportRows AS (
    SELECT
        FACILITY,
        MATCH_STATUS,
        GL,

        LEGACY_CATEGORY,
        NEW_CATEGORY,

        LEGACY_DESCRIPTION,
        NEW_DESCRIPTION,

        LEGACY_COST,
        NEW_COST,
        DIFFERENCE

    FROM GLComparison


    UNION ALL


    SELECT
        FACILITY,

        'LEGACY_NO_GL'
            AS MATCH_STATUS,

        NULL AS GL,

        LEGACY_CATEGORY,

        NULL AS NEW_CATEGORY,

        NULL AS LEGACY_DESCRIPTION,

        NULL AS NEW_DESCRIPTION,

        LEGACY_COST,

        0 AS NEW_COST,

        -LEGACY_COST
            AS DIFFERENCE

    FROM LegacyNoGL
),


/* ============================================================
   ADD REPORT TOTALS BEFORE FILTERING OUT EXACT MATCHES
   ============================================================ */

ReportWithTotals AS (
    SELECT
        *,

        SUM(LEGACY_COST)
            OVER (PARTITION BY FACILITY)
            AS FACILITY_LEGACY_COST,

        SUM(NEW_COST)
            OVER (PARTITION BY FACILITY)
            AS FACILITY_NEW_COST,

        SUM(DIFFERENCE)
            OVER (PARTITION BY FACILITY)
            AS FACILITY_DIFFERENCE,

        SUM(LEGACY_COST)
            OVER ()
            AS TOTAL_LEGACY_COST,

        SUM(NEW_COST)
            OVER ()
            AS TOTAL_NEW_COST,

        SUM(DIFFERENCE)
            OVER ()
            AS TOTAL_DIFFERENCE

    FROM ReportRows
)


/* ============================================================
   FINAL DIFFERENCE REPORT

   Exact matches are suppressed.
   ============================================================ */

SELECT
    FACILITY,

    MATCH_STATUS,

    GL,

    LEGACY_CATEGORY,
    NEW_CATEGORY,

    LEGACY_DESCRIPTION,
    NEW_DESCRIPTION,

    LEGACY_COST,
    NEW_COST,

    DIFFERENCE,

    CAST(
        CASE
            WHEN LEGACY_COST = 0
                THEN NULL

            ELSE
                100.0 * DIFFERENCE
                / ABS(LEGACY_COST)
        END

        AS decimal(12,2)
    ) AS DIFFERENCE_PCT,

    FACILITY_LEGACY_COST,
    FACILITY_NEW_COST,
    FACILITY_DIFFERENCE,

    TOTAL_LEGACY_COST,
    TOTAL_NEW_COST,
    TOTAL_DIFFERENCE,

    CAST(
        100.0 * TOTAL_DIFFERENCE
        / NULLIF(ABS(TOTAL_LEGACY_COST), 0)

        AS decimal(12,2)
    ) AS TOTAL_DIFFERENCE_PCT

FROM ReportWithTotals

WHERE
       ABS(DIFFERENCE) > 0.01

    OR ISNULL(LEGACY_CATEGORY, '')
        <> ISNULL(NEW_CATEGORY, '')

ORDER BY
    ABS(FACILITY_DIFFERENCE) DESC,
    FACILITY,
    ABS(DIFFERENCE) DESC;
