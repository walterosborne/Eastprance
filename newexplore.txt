Promise.all([
  ['Raw', '/api/controllable-costs-databricks-raw'],
  ['New', '/api/controllable-costs-databricks-new?refresh=1']
].map(async ([name, url]) => {
  const response = await fetch(url);
  const data = await response.json();
  return {
    name,
    status: response.status,
    table: data.tableName,
    sourceRows: data.sourceRowCount,
    unmatchedHierarchy: data.hierarchyUnmatchedRowCount,
    excludedByScope: data.excludedBySqlScopeRowCount,
    finalRows: data.rowCount,
    totalCost: data.totalCost,
    error: data.error
  };
})).then(console.table);
