SELECT
    "FY_PD_NO_CD",

    CASE
        WHEN UPPER("ACCT_NAME") LIKE '%PROPERTY TAX%'
          OR UPPER("L3_ACCT_NAME") LIKE '%PROPERTY TAX%'
            THEN 'Property Taxes'

        WHEN UPPER("ACCT_NAME") LIKE '%UTILIT%'
          OR UPPER("L3_ACCT_NAME") LIKE '%UTILIT%'
          OR UPPER("ACCT_NAME") LIKE '%LIGHT/POWER%'
          OR UPPER("ACCT_NAME") LIKE '%HEATING FUEL%'
            THEN 'Utilities'

        WHEN UPPER("ACCT_NAME") LIKE '%MAINT%'
          OR UPPER("L3_ACCT_NAME") LIKE '%MAINT%'
          OR UPPER("ACCT_NAME") LIKE '%REPAIR%'
          OR UPPER("L3_ACCT_NAME") LIKE '%REPAIR%'
            THEN 'Maintenance & Repairs'

        WHEN UPPER("ACCT_NAME") LIKE '%INSUR%'
          OR UPPER("L3_ACCT_NAME") LIKE '%INSUR%'
            THEN 'Insurance'

        WHEN UPPER("ACCT_NAME") LIKE '%LEASE EXP%'
          OR UPPER("L3_ACCT_NAME") LIKE '%LEASE EXP%'
          OR UPPER("ACCT_NAME") LIKE '%RENT%'
          OR UPPER("L3_ACCT_NAME") LIKE '%RENT%'
            THEN 'Rent - Land & Building'

        WHEN UPPER("ACCT_NAME") LIKE '%DEPR%'
          OR UPPER("L3_ACCT_NAME") LIKE '%DEPR%'
          OR UPPER("ACCT_NAME") LIKE '%LEASEHOLD IMPROV%'
          OR UPPER("L3_ACCT_NAME") LIKE '%LEASEHOLD IMPROV%'
            THEN 'Depreciation & LHI'

        ELSE 'Other'
    END AS "COST_CATEGORY",

    COUNT(*) AS "ROW_COUNT",
    SUM("AMT") AS "AMT"

FROM "_SYS_BIC"."ngc.DWH.SectorContent.NGSS.SP_COSTPOINT.Restricted/CVG_CP_GL_DETL_RESTRICTED"
(
    'PLACEHOLDER' = ('$$IP_FY_PD_NO_TO$$', '202608'),
    'PLACEHOLDER' = ('$$IP_ACCT_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_ORG_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_L1_PROJ_SEG_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_FY_PD_NO_FROM$$', '202601'),
    'PLACEHOLDER' = ('$$IP_INCL_ITD_PD_NO$$', 'Y'),
    'PLACEHOLDER' = ('$$IP_INCL_ALL_HDR$$', 'N')
)

WHERE "CP_INSTANCE" = 'CPENT'
  AND "L1_ORG_NAME" = 'MSG'
  AND "L3_ORG_NAME" IN (
      'ARMAMENT SYSTEMS',
      'Advanced Weapons',
      'ROCKET CENTER (ABL)',
      'ELKTON',
      'NRE',
      'PROPULSION SYSTEMS',
      'WS TPS'
  )

  AND (
      UPPER("ACCT_NAME") LIKE '%PROPERTY TAX%'
      OR UPPER("L3_ACCT_NAME") LIKE '%PROPERTY TAX%'
      OR UPPER("ACCT_NAME") LIKE '%UTILIT%'
      OR UPPER("L3_ACCT_NAME") LIKE '%UTILIT%'
      OR UPPER("ACCT_NAME") LIKE '%LIGHT/POWER%'
      OR UPPER("ACCT_NAME") LIKE '%HEATING FUEL%'
      OR UPPER("ACCT_NAME") LIKE '%MAINT%'
      OR UPPER("L3_ACCT_NAME") LIKE '%MAINT%'
      OR UPPER("ACCT_NAME") LIKE '%REPAIR%'
      OR UPPER("L3_ACCT_NAME") LIKE '%REPAIR%'
      OR UPPER("ACCT_NAME") LIKE '%INSUR%'
      OR UPPER("L3_ACCT_NAME") LIKE '%INSUR%'
      OR UPPER("ACCT_NAME") LIKE '%LEASE EXP%'
      OR UPPER("L3_ACCT_NAME") LIKE '%LEASE EXP%'
      OR UPPER("ACCT_NAME") LIKE '%RENT%'
      OR UPPER("L3_ACCT_NAME") LIKE '%RENT%'
      OR UPPER("ACCT_NAME") LIKE '%DEPR%'
      OR UPPER("L3_ACCT_NAME") LIKE '%DEPR%'
      OR UPPER("ACCT_NAME") LIKE '%LEASEHOLD IMPROV%'
      OR UPPER("L3_ACCT_NAME") LIKE '%LEASEHOLD IMPROV%'
  )

GROUP BY
    "FY_PD_NO_CD",
    CASE
        WHEN UPPER("ACCT_NAME") LIKE '%PROPERTY TAX%'
          OR UPPER("L3_ACCT_NAME") LIKE '%PROPERTY TAX%'
            THEN 'Property Taxes'
        WHEN UPPER("ACCT_NAME") LIKE '%UTILIT%'
          OR UPPER("L3_ACCT_NAME") LIKE '%UTILIT%'
          OR UPPER("ACCT_NAME") LIKE '%LIGHT/POWER%'
          OR UPPER("ACCT_NAME") LIKE '%HEATING FUEL%'
            THEN 'Utilities'
        WHEN UPPER("ACCT_NAME") LIKE '%MAINT%'
          OR UPPER("L3_ACCT_NAME") LIKE '%MAINT%'
          OR UPPER("ACCT_NAME") LIKE '%REPAIR%'
          OR UPPER("L3_ACCT_NAME") LIKE '%REPAIR%'
            THEN 'Maintenance & Repairs'
        WHEN UPPER("ACCT_NAME") LIKE '%INSUR%'
          OR UPPER("L3_ACCT_NAME") LIKE '%INSUR%'
            THEN 'Insurance'
        WHEN UPPER("ACCT_NAME") LIKE '%LEASE EXP%'
          OR UPPER("L3_ACCT_NAME") LIKE '%LEASE EXP%'
          OR UPPER("ACCT_NAME") LIKE '%RENT%'
          OR UPPER("L3_ACCT_NAME") LIKE '%RENT%'
            THEN 'Rent - Land & Building'
        WHEN UPPER("ACCT_NAME") LIKE '%DEPR%'
          OR UPPER("L3_ACCT_NAME") LIKE '%DEPR%'
          OR UPPER("ACCT_NAME") LIKE '%LEASEHOLD IMPROV%'
          OR UPPER("L3_ACCT_NAME") LIKE '%LEASEHOLD IMPROV%'
            THEN 'Depreciation & LHI'
        ELSE 'Other'
    END

ORDER BY
    "FY_PD_NO_CD",
    "COST_CATEGORY";
