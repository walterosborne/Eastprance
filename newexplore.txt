SELECT
    "ACCT_ID",
    "ACCT_NAME",
    "L1_ACCT_NAME",
    "L2_ACCT_NAME",
    "L3_ACCT_NAME",
    COUNT(*) AS "ROW_COUNT",
    SUM("AMT") AS "Q1_AMT"

FROM "_SYS_BIC"."ngc.DWH.SectorContent.NGSS.SP_COSTPOINT.Restricted/CVG_CP_GL_DETL_RESTRICTED"
(
    'PLACEHOLDER' = ('$$IP_FY_PD_NO_TO$$', '202603'),
    'PLACEHOLDER' = ('$$IP_ACCT_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_ORG_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_L1_PROJ_SEG_ID$$', '''*'''),
    'PLACEHOLDER' = ('$$IP_FY_PD_NO_FROM$$', '202601'),
    'PLACEHOLDER' = ('$$IP_INCL_ITD_PD_NO$$', 'Y'),
    'PLACEHOLDER' = ('$$IP_INCL_ALL_HDR$$', 'N')
)

WHERE "CP_INSTANCE" = 'CPENT'
  AND "L1_ORG_NAME" = 'MSG'
  AND "L3_ORG_NAME" = 'ROCKET CENTER (ABL)'
  AND (
       UPPER("ACCT_NAME") LIKE '%RENT%'
    OR UPPER("ACCT_NAME") LIKE '%LEASE%'
    OR UPPER("ACCT_NAME") LIKE '%DEPR%'
    OR UPPER("ACCT_NAME") LIKE '%PROPERTY TAX%'
    OR UPPER("ACCT_NAME") LIKE '%INSUR%'
    OR UPPER("ACCT_NAME") LIKE '%MAINT%'
    OR UPPER("ACCT_NAME") LIKE '%REPAIR%'
    OR UPPER("ACCT_NAME") LIKE '%UTILIT%'
    OR UPPER("ACCT_NAME") LIKE '%ELECTRIC%'
    OR UPPER("ACCT_NAME") LIKE '%POWER%'
    OR UPPER("ACCT_NAME") LIKE '%FUEL%'
    OR UPPER("ACCT_NAME") LIKE '%CUSTOD%'
    OR UPPER("ACCT_NAME") LIKE '%GROUND%'
  )

GROUP BY
    "ACCT_ID",
    "ACCT_NAME",
    "L1_ACCT_NAME",
    "L2_ACCT_NAME",
    "L3_ACCT_NAME"

ORDER BY "Q1_AMT" DESC;
