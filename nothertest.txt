SELECT
    TRY_CONVERT(INT, GJAHR) AS [year],
    TRY_CONVERT(INT, POPER) AS [month],
    Cost_Type,
    COUNT(*) AS row_count,
    SUM(TRY_CONVERT(DECIMAL(18,2), KSL)) AS cost
FROM dbo.rb_Actuals_SAP_Transformation
WHERE TRY_CONVERT(INT, POPER) BETWEEN 1 AND 12
GROUP BY
    TRY_CONVERT(INT, GJAHR),
    TRY_CONVERT(INT, POPER),
    Cost_Type
ORDER BY
    [year] DESC,
    [month] DESC,
    Cost_Type;
